## 题意
给定一个长为 $n$ 的字符串，$m$ 次询问 $s[a..b]$ 的所有子串和 $s[c..d]$ 的最长公共前缀的最大值。
$$
1 \le n, m \le 10^5
$$
## 思路
二分答案，转为判定 $s[c..c+mid-1]$ 是否在 $s[a..b]$ 中出现过。建出 SAM，倍增跳父亲找到 $s[c..c+mid-1]$ 对应的节点 $u$，相当于是判断 $u$ 的 endpos 是否在 $a+mid-1 \sim b$ 出现过。在 parent 树上，$u$ 的 endpos 是 $u$ 的儿子的 endpos 的并。这个直接用线段树合并维护即可。时间复杂度 $O(n \log n + q \log^2 n)$。

## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int n, m, tot = 1, lst = 1, idx, ch[N][26], fa[N];
int lc[N * 40], rc[N * 40], len[N], rt[N], pos[N], cnt[N], ord[N], st[N][22];
char s[N];
inline int add(int p, int l, int r) {
	int x = ++ idx;
	if (l == r) return x;
	int mid = l + r >> 1; 
	if (p <= mid) lc[x] = add(p, l, mid);
	else rc[x] = add(p, mid + 1, r);
	return x;
}
inline int merge(int x, int y, int l, int r) {
	if (!x || !y) return x | y;
	int p = ++ idx;
	if (l == r) return p;
	int mid = l + r >> 1;
	lc[p] = merge(lc[x], lc[y], l, mid);
	rc[p] = merge(rc[x], rc[y], mid + 1, r);
	return p;
}
inline bool ask(int p, int l, int r, int L, int R) {
	if (!p) return 0;
	if (L >= l && R <= r) return 1;
	int mid = L + R >> 1;
	if (l <= mid && ask(lc[p], l, r, L, mid)) return 1;
	if (r > mid && ask(rc[p], l, r, mid + 1, R)) return 1;
	return 0;
}
inline void ins(int c, int x) {
	int p = lst, u = ++ tot;
	len[u] = len[p] + 1; pos[x] = lst = u;
	rt[u] = add(x, 1, n);
	while (p && !ch[p][c]) ch[p][c] = u, p = fa[p];
	if (!p) fa[u] = 1;
	else {
		int q = ch[p][c];
		if (len[q] == len[p] + 1) fa[u] = q;
		else {
			int nw = ++ tot;
			memcpy(ch[nw], ch[q], sizeof(ch[q]));
			fa[nw] = fa[q], len[nw] = len[p] + 1;
			fa[u] = fa[q] = nw;
			while (p && ch[p][c] == q) ch[p][c] = nw, p = fa[p];
		}
	}
}
inline bool check(int a, int b, int c, int mid) {
	int u = pos[c + mid - 1];
	for (int j = 17; j >= 0; j --)
		if (len[st[u][j]] >= mid) u = st[u][j];
	return ask(rt[u], a + mid - 1, b, 1, n);
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> m;
	for (int i = 1; i <= n; i ++) cin >> s[i], ins(s[i] - 'a', i);
	for (int i = 1; i <= tot; i ++) cnt[len[i]] ++, st[i][0] = fa[i];
	for (int i = 1; i <= n; i ++) cnt[i] += cnt[i - 1];
	for (int i = 1; i <= tot; i ++) ord[cnt[len[i]] --] = i;
	for (int i = 1; (1 << i) <= tot; i ++)
		for (int j = 1; j <= tot; j ++)
			st[j][i] = st[st[j][i - 1]][i - 1];
	for (int i = tot; i >= 1; i --) {
		int u = ord[i];
		rt[fa[u]] = merge(rt[fa[u]], rt[u], 1, tot); 
	}
	while (m --) {
		int a, b, c, d;
		cin >> a >> b >> c >> d;
		int l = 1, r = min(b - a + 1, d - c + 1), ans = 0;
		while (l <= r) {
			int mid = l + r >> 1;
			if (check(a, b, c, mid)) l = mid + 1, ans = mid;
			else r = mid - 1;
		}
		cout << ans << '\n';
	}
	return 0;
}
```
