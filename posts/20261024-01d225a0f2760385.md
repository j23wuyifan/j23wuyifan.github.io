## 题意
给定一个字符串 $S$，有 $Q$ 次询问。

每次给出字符串 $T$ 和区间 $[l,r]$，求 **$T$ 中有多少个本质不同的非空子串，没有在 $S[l\ldots r]$ 中出现过**。

其中，本质不同指字符串内容不同，相同子串出现多次只算一次。

数据范围：$|S|\le 5\times10^5$，$Q\le10^5$，$\sum |T|\le10^6$。
## 思路
首先肯定要对 $T$ 建出 SAM，对于 SAM 上的一个节点，它没在 $S[l \dots r]$ 出现的一定是一些长度 $> k$ 的字符串。那么节点 $u$ 的贡献就是 $\max(0, len[u] - max(len[fa[u]], k)$。接下来只要求出 $k$ 即可。遍历一遍 $T$，求出 $f_i$ 表示 $f_i$ 是最长的 $l$ 使得 $T[i-l+1 \dots i]$ 在 $S$ 中出现。$k$ 就是 $u$ 的任意一个 endpos 的 $f_i$。$f_i$ 每次最多加 $1$，总变化量是 $O(|T|)$ 的，检查有没有在 $S[l \dots r]$ 出现就类似 [HEOI2016/TJOI2016] 字符串，用线段树合并维护 endpos 集合就好了。时间复杂度单 $\log$。

## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e6 + 10;
int n, m, lc[N * 40], rc[N * 40], idx, rt[N], pos[N], st[N][22];
string s;
int ch[N][27], len[N], fa[N], ord[N], cnt[N], tot = 1, lst = 1;
inline int add(int p, int l, int r) {
	int x = ++ idx;
	if (l == r) return x;
	int mid = l + r >> 1;
	if (p <= mid) lc[x] = add(p, l, mid);
	else rc[x] = add(p, mid + 1, r);
	return x;
}
inline int merge(int x, int y, int l, int r) {
	if (!x || !y) return x | y;
	int p = ++ idx;
	int mid = l + r >> 1;
	lc[p] = merge(lc[x], lc[y], l, mid);
	rc[p] = merge(rc[x], rc[y], mid + 1, r);
	return p;
}
inline bool ask(int p, int l, int r, int L, int R) {
	if (!p) return 0;
	if (L >= l && R <= r) return 1;
	int mid = L + R >> 1;
	if (l <= mid && ask(lc[p], l, r, L, mid)) return 1;
	if (r > mid && ask(rc[p], l, r, mid + 1, R)) return 1;
	return 0;
}
inline void ins(int c, int x) {
	int u = ++ tot, p = lst;
	pos[x] = lst = u, len[u] = len[p] + 1;
	rt[u] = add(x, 1, n);
	while (p && !ch[p][c]) ch[p][c] = u, p = fa[p];
	if (!p) fa[u] = 1;
	else {
		int q = ch[p][c];
		if (len[q] == len[p] + 1) fa[u] = q;
		else {
			int nw = ++ tot;
			memcpy(ch[nw], ch[q], sizeof(ch[q]));
			fa[nw] = fa[q], len[nw] = len[p] + 1;
			fa[u] = fa[q] = nw;
			while (p && ch[p][c] == q) ch[p][c] = nw, p = fa[p];
		}
	}
}
struct SAM {
	int ch[N][27], len[N], fa[N], tot = 1, lst = 1, pos[N];
	inline void ins(int c, int x) {
		int u = ++ tot, p = lst;
		pos[u] = x;
		lst = u, len[u] = len[p] + 1;
		while (p && !ch[p][c]) ch[p][c] = u, p = fa[p];
		if (!p) fa[u] = 1;
		else {
			int q = ch[p][c];
			if (len[q] == len[p] + 1) fa[u] = q;
			else {
				int nw = ++ tot;
				memcpy(ch[nw], ch[q], sizeof(ch[q]));
				fa[nw] = fa[q], len[nw] = len[p] + 1, pos[nw] = x;
				fa[u] = fa[q] = nw;
				while (p && ch[p][c] == q) ch[p][c] = nw, p = fa[p];
			}
		}
	}
	inline void init() {
		for (int i = 1; i <= tot; i ++) {
			memset(ch[i], 0, sizeof(ch[i]));
			len[i] = fa[i] = pos[i] = cnt[i] = 0;
		}
		tot = lst = 1;
	}
} tr;
int ql, qr, f[N], p = 0;
inline bool chk(int u, int k) {
	return ask(rt[u], ql + k - 1, qr, 1, n);
}
inline void solve() {
	string t;
	cin >> t;
	int L = t.size();
	t = ' ' + t;
	cin >> ql >> qr;
	tr.init();
	for (int i = 1; i <= L; i ++) tr.ins(t[i] - 'a', i);
	p = 1;
	for (int i = 1, d = 0; i <= L; i ++) {
		const int c = t[i] - 'a';
//		cout << t[i] << ' ' << p << ' ' << ch[p][c] << ' ';
		if (ch[p][c] && chk(ch[p][c], d + 1)) d ++, p = ch[p][c];
		else {
			while (d) {
				d --;
				if (len[fa[p]] == d) p = fa[p];
				if (ch[p][c] && chk(ch[p][c], d + 1)) {
					d ++, p = ch[p][c];
					break;
				}
			}
		}
		f[i] = d;
//		cout << f[i] << "   ";
	}
//	cout << '\n';
	long long ans = 0;
	for (int i = 2; i <= tr.tot; i ++)
		ans += max(0, tr.len[i] - max(tr.len[tr.fa[i]], f[tr.pos[i]]));
	cout << ans << '\n';
}
signed main() {
	cin >> s;
	n = s.size(), s = ' ' + s;
	for (int i = 1; i <= n; i ++) ins(s[i] - 'a', i);
	for (int i = 1; i <= tot; i ++) cnt[len[i]] ++;
	for (int i = 1; i <= n; i ++) cnt[i] += cnt[i - 1];
	for (int i = 1; i <= tot; i ++) ord[cnt[len[i]] --] = i;
	for (int i = tot; i >= 1; i --) {
		int u = ord[i];
		rt[fa[u]] = merge(rt[fa[u]], rt[u], 1, n);
	}
	cin >> m;
	while (m --) solve();
	return 0;
}
```
