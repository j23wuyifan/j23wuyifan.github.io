## 题意
给定 $n$ 和序列 $a_{0\sim 2^n-1}$，$q$ 次询问，每次给出一个大小为 $k$ 的可重集 $S$，求：

$$\sum_{T\sube S}a_{\bigoplus_{x\in T}x}$$

答案对 $998244353$ 取模。

- $1\le n\le 20$；
- $1\le q\le 2\times 10^5$；
- $1\le k,\sum k\le 4\times 10^6$；
## 思路
设 $S$ 的线性基秩为 $r$，张成空间为 $V$。每个 $x\in V$ 都被恰好 $2^{k-r}$ 个子集异或得到，因此答案为
$$
2^{k-r}\sum_{x\in V}a_x
$$
直接枚举复杂度 $O(2^r)$。

记二进制内积为：
$$
\langle x,y\rangle=\operatorname{popcount}(x\&y)\bmod2.
$$
预处理异或 FWT：
$$
b_y=\sum_x(-1)^{\langle x,y\rangle}a_x.
$$

FWT 的逆变换是：
$$
a_x=\frac{1}{2^n}\sum_y (-1)^{\langle x,y\rangle}b_y
$$
利用逆变换，有
$$
\begin{aligned}
  \sum_{x\in V}a_x &= \sum_{x \in v}\frac{1}{2^n}\sum_y (-1)^{\langle x,y\rangle}b_y \\
                 &= \frac{1}{2^n}\sum_y b_y \sum_{x \in V} (-1)^{\langle x,y\rangle} \\
                 &= \frac{2^r}{2^n}\sum_{y\in V^\perp}b_y
\end{aligned}
$$

其中 $V^\perp$ 表示与 $V$ 中所有向量的二进制内积均为 $0$ 的向量空间，维数为 $n-r$。因为对于 $y\notin V^\perp$，$\sum_{y\in V^\perp}b_y=0$，因为贡献是一对一对的。否则对于 $y \in V^\perp$，$\sum_{y\in V^\perp}b_y=2^r$。直接枚举 $y \in V^\perp$ 的复杂度是 $O(2^{n-r})$ 的。

剩下的就是如何求出 $V^\perp$。考虑求出 $n-r$ 个基底。因为 $(a\oplus b) \& c=(a \& c) \oplus (b \& c)$。下面这个就是对的，因为对于 $p_j=0$ 的 $j$ 只有一个 $x$ 第 $j$ 位为 $1$。一定构成一个基。
``` cpp
for (int j = 0; j < n; j ++)
	if (!p[j]) {
		int x = 1 << j;
		for (int i = 0; i < n; i ++)
			if (p[i] && __builtin_parity((unsigned)(p[i] & x)))
				x |= 1 << i;
		c[++ cnt] = x;
	}
```
复杂度 $O(n^2)$。

两个复杂度平衡一下就是 $O(q(2^\frac{n}{2}+n^2)+n \sum k)$。

## 代码

```cpp
#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int N = (1 << 20) + 10, M = 4e6 + 10, mod = 998244353;
inline void add(int &x, int y) {
	x = x + y >= mod ? x + y - mod : x + y;
}
inline int qpow(int a, int b) {
	int ans = 1;
	while (b) {
		if (b & 1) ans = 1ll * ans * a % mod;
		a = 1ll * a * a % mod, b >>= 1;
	}
	return ans;
}
int n, m, k, a[N], b[N], q, pw2[M], p[21], pcnt[N], c[21], v[N];
inline void solve() {
	cin >> k;
	int r = 0, cnt = 0;
	memset(p, 0, sizeof(p));
	for (int i = 1, x; i <= k; i ++) {
		cin >> x;
		for (int j = m - 1; j >= 0; j --) {
			if (x >> j & 1) {
				if (!p[j]) {
					p[j] = x, r ++;
					break;
				}
				x ^= p[j];
			}
		}
	}
	int *f, mul;
	if (r * 2 <= m) {
		for (int i = 0; i < m; i ++)
			if (p[i]) c[++ cnt] = p[i];
		f = a, mul = pw2[k - r];
	} else {
		for (int i = 0; i < m; i ++) {
			if (!p[i]) {
				int x = 1 << i;
				for (int j = 0; j < m; j ++)
					if (p[j] && pcnt[x & p[j]]) x |= 1 << j;
				c[++ cnt] = x;
			}
		}
		f = b;
		mul = 1ll * pw2[k] * qpow(n, mod - 2) % mod;
	}
	int tot = 1;
	v[1] = 0;
	ll ans = f[0];
	for (int i = 1; i <= cnt; i ++) {
		for (int j = 1; j <= tot; j ++) {
			v[tot + j] = v[j] ^ c[i];
			ans += f[v[tot + j]];
		}
		tot <<= 1;
	}
	cout << 1ll * ans % mod * mul % mod << '\n';
}
signed main() {
    ios :: sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
	pw2[0] = 1;
	for (int i = 1; i <= M - 10; i ++) pw2[i] = pw2[i - 1] * 2 % mod;
	for (int i = 1; i < (1 << 20); i ++) pcnt[i] = pcnt[i >> 1] ^ (i & 1);
	cin >> m >> q;
	n = 1 << m;
	for (int i = 0; i < n; i ++) cin >> a[i], b[i] = a[i];
	for (int mid = 2; mid <= n; mid <<= 1) {
		for (int j = 0; j < n; j += mid)
			for (int k = 0; k < mid / 2; k ++) {
				int x = b[j + k], y = b[j + k + mid / 2];
				b[j + k] = (x + y) % mod;
				b[j + k + mid / 2] = (x + mod - y) % mod;
			}
	}
	while (q --) solve();
	return 0;
}
```
