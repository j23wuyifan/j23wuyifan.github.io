## 题意
给定一个 $d$ 层的有向图，边只从第 $i$ 层连向第 $i+1$ 层，重边只算一次。

每个点 $u$ 对应一个多项式 $F_u(x)$。第一层的多项式已知，次数不超过 $2$；其余点满足
$$
F_u(x)=\sum_{v\to u}\bigl(F_v(x^2)\bigr)^2.
$$
每次询问给出 $u,k$，求 $F_u(k)\bmod7340033$。
## 思路
令 $P=7340033$，$M=P-1$。
$3$ 是模 $P$ 的原根，因此每个非零 $k$ 都能唯一写成 $3^t\bmod P$。预处理离散对数表 $\operatorname{lg}[3^t]=t$。

对于第 $i$ 层，令
$$
L_i=\frac{M}{2^{i-1}},\qquad A_u[t]=F_u(3^t)
$$
每次查询的答案就是 $F_u[k]=A_u[\operatorname{lg}[k]]$。

由于第 $i$ 层多项式的所有指数都是 $2^{i-1}$ 的倍数，故
$$
F_u(3^{t+L_i})=F_u(3^t)，\qquad A_u[t]=A_u[t+L_i]
$$
所以只要保存长度为 $L_i$ 的 $A$ 数组即可。
由题目递推式，
$$
A_v[t]=\sum_{u\to v}A_u[2t]^2\pmod P
$$
逐层处理，直接转移。答案就是 $A_u[\operatorname{lg}[k] \mod L_i]$。

每层的复杂度是 $O(n_i L_i)$。因为 $n_i <= 2 n_{i - 1}$，所以 $n_i L_i \le n_1 L_1=O(n_1 P)$。总复杂度是 $O(n_1dP + Q)$。
## 代码

```cpp
#include<bits/stdc++.h>
#define ll long long
#define int unsigned int
using namespace std;
const int mod = 7340033, M = mod - 1;
int d, n[12], a[6][3], lg[mod];
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> d;
	for (int i = 1; i <= d; i ++) cin >> n[i];
	for (int i = 1, p; i <= n[1]; i ++) {
		cin >> p;
		for (int j = 0; j <= p; j ++) cin >> a[i][j];
	}
	vector<vector<int> > f(n[1] + 1, vector<int> (M));
	for (int j = 0, x = 1; j < M; j ++, x = 1ll * 3 * x % mod) {
		lg[x] = j;
		for (int u = 1; u <= n[1]; u ++)
			f[u][j] = ((1ll * a[u][2] * x + a[u][1]) % mod * x + a[u][0]) % mod;
	}
	int len = M;
	for (int i = 1, m, q; i <= d; i ++) {
		cin >> m >> q;
		vector<vector<int> > e(n[i + 1] + 1);
		for (int j = 1, u, v; j <= m; j ++) {
			cin >> u >> v;
			e[v].push_back(u);
		}
		unsigned ans = 0;
		for (int j = 1, u, k; j <= q; j ++) {
			cin >> u >> k;
			// ans ^= (1u * (q - j) * f[u][lg[k] % len]);
			 ans ^= uint32_t(q - j) * uint32_t(f[u][lg[k] % len]);
		}
		cout << ans * uint32_t(i) << '\n';
		if (i == d) break;
		len >>= 1;
		for (int u = 1; u <= n[i]; u ++)
			for (int j = 0; j < len; j ++)
				f[u][j] = 1ll * f[u][j << 1] * f[u][j << 1] % mod;
		vector<vector<int> > g(n[i + 1] + 1);
		for (int v = 1; v <= n[i + 1]; v ++) {
			sort (e[v].begin(), e[v].end());
			e[v].erase(unique(e[v].begin(), e[v].end()), e[v].end());
			g[v].resize(len);
			for (int u : e[v]) {
				for (int j = 0; j < len; j ++) {
					int x = g[v][j] + f[u][j];
					g[v][j] = x >= mod ? x - mod : x;
				}
			}
		}
		f.swap(g);
	}
	return 0;
}
```
