## 题意
给定 $n,m$，你需要维护一个 $[1,n)$ 的数轴上区间的初始为空的可重集合，支持三种操作共 $m$ 次：

1. 插入一个区间 $[l,r)$。
2. 删除第 $t$ 次操作插入的区间。
3. 给出一个区间 $[l,r)$，判断当前可重集合是否存在一个子集，使得子集中所有区间的并恰好是 $[l,r)$。
## 思路
将开区间变成闭区间。对于所有 $\subseteq [l, r]$ 的区间肯定要选。

分治计算跨过 $mid$ 的查询，对询问产生影响的有 $2$ 种区间，跨过 $mid$ 的和在 $mid$ 两边的。跨过 $mid$ 的区间都有交，并集一定是一个区间，考虑维护出 $f_i,g_i$ 表示第 $i$ 个询问的并集。这个就按顺序执行操作，维护 
$$
v_x=\min\{r_i\mid l_i=x，r_i > mid\}
$$
$f_i$ 就是最左边的一个 $x$ 满足 $v_x \le r_i$。$g_i$ 用同样的方法维护。这个可以用一颗线段树快速维护出来。

对于在 $mid$ 左边的区间，维护 $h_t$ 表示 $t$ 这个询问从 $l_t$ 开始第一个未覆盖的下标。

考虑扫描线，当扫到下标 $x$ 时，满足 $l_t=x$ 的 $h_t$ 设为 $x$。

对每个左端点为 $x$、存活时间为 $[s,e]$ 的区间 $[x,y]$，执行 
$$
h_t\gets\max(h_t,y+1),\qquad t\in[s,e].
$$

取出所有 $h_t=x$ 的询问，记录其最终覆盖位置并删除，它们在 $x$ 处断开。若 $h_t < f_t$，$t$ 的答案为 $N$。

时间复杂度为 $O(n \log^2 n)$。真难写啊。
## 代码

```cpp
#include<bits/stdc++.h>
#define lc id << 1
#define rc id << 1 | 1
using namespace std;
const int N = 1e6 + 10, inf = 1e9;
int n, m, ed[N], tp[N], l[N], r[N], ans[N], f[N];
vector<multiset<int> > s;
struct MinTree {
	vector<int> v;
	MinTree (int k) : v(k * 4 + 4, inf) {}
	inline void pushup(int id) {
		v[id] = min(v[lc], v[rc]);
	}
	inline void upd(int id, int p, int x, int l, int r) {
		if (l == r) {
			v[id] = x;
			return;
		}
		int mid = l + r >> 1;
		if (p <= mid) upd(lc, p, x, l, mid);
		else upd(rc, p, x, mid + 1, r);
		pushup(id);
	}
	inline int qry(int id, int l, int r, int x, int y) {
		if (r < x || v[id] > y) return 0;
		if (l == r) return l;
		int mid = l + r >> 1;
		int res = qry(lc, l, mid, x, y);
		if (res) return res;
		return qry(rc, mid + 1, r, x, y);
	}
};
struct Tree {
	vector<int> v, tag;
	Tree(int k) : v(k * 4 + 4, inf), tag(k * 4 + 4) {}
	inline void put(int id, int x) {
		v[id] = max(v[id], x);
		tag[id] = max(tag[id], x);
	}
	inline void pushdown(int id) {
		if (tag[id]) {
			put(lc, tag[id]);
			put(rc, tag[id]);
			tag[id] = 0;
		}
	}
	inline void pushup(int id) {
		v[id] = min(v[lc], v[rc]);
	}
	inline void upd(int id, int p, int x, int l, int r) {
		if (l == r) {
			v[id] = x;
			tag[id] = 0;
			return;
		}
		pushdown(id);
		int mid = l + r >> 1;
		if (p <= mid) upd(lc, p, x, l, mid);
		else upd(rc, p, x, mid + 1, r);
		pushup(id);
	}
	inline void add(int id, int l, int r, int x, int L, int R) {
		if (L >= l && R <= r) {
			put(id, x);
			return;
		}
		pushdown(id);
		int mid = L + R >> 1;
		if (l <= mid) add(lc, l, r, x, L, mid);
		if (r > mid) add(rc, l, r, x, mid + 1, R);
		pushup(id);
	}
	inline int pop(int id, int l, int r) {
		if (l == r) {
			v[id] = inf;
			tag[id] = 0;
			return l;
		}
		pushdown(id);
		int mid = l + r >> 1, x;
		if (v[lc] <= v[rc]) x = pop(lc, l, mid);
		else x = pop(rc, mid + 1, r);
		pushup(id);
		return x;
	}
};
inline int L(int i, int d) {
	return d ? n - r[i] + 1 : l[i];
}
inline int R(int i, int d) {
	return d ? n - l[i] + 1 : r[i];
}
inline void work(int a, int b, int mid, const vector<int> &c, const vector<int> &h,
                 const vector<int> &q, int d) {
	int k = q.size();
	if (!k) return;
	if (d) {
		int x = a;
		a = n - b + 1;
		b = n - x + 1;
		mid = n - mid;
	}
	vector<pair<int, int> > ev;
	for (int i : c) {
		ev.emplace_back(i, i);
		ev.emplace_back(ed[i] + 1, -i);
	}
	for (int i : q) {
		ev.emplace_back(i, 0);
		f[i] = mid + 1;
	}
	sort (ev.begin(), ev.end());
	{
		MinTree tr(b - a + 1);
		for (auto [t, id] : ev) {
			if (!id) {
				int x = tr.qry(1, a, b, L(t, d), R(t, d));
				if (x) f[t] = x;
				continue;
			}
			int i = abs(id), x = L(i, d), y = R(i, d);
			if (id > 0) s[x].insert(y);
			else s[x].erase(s[x].find(y));
			tr.upd(1, x, s[x].size() ? *s[x].begin() : inf, a, b);
		}
	}
	ev.clear();
	for (int i : h) ev.emplace_back(L(i, d), i);
	for (int i = 1; i <= k; i ++) ev.emplace_back(L(q[i - 1], d), -i);
	ev.emplace_back(inf, 0);
	sort (ev.begin(), ev.end());
	Tree tr(k);
	for (auto [x, id] : ev) {
		while (tr.v[1] < x) {
			int z = tr.v[1], i = tr.pop(1, 1, k);
			if (z < f[q[i - 1]]) ans[q[i - 1]] = 0;
		}
		if (!id) break;
		if (id < 0) tr.upd(1, -id, x, 1, k);
		else {
			int u = lower_bound(q.begin(), q.end(), id) - q.begin() + 1;
			int v = upper_bound(q.begin(), q.end(), ed[id]) - q.begin();
			if (u <= v) tr.add(1, u, v, R(id, d) + 1, 1, k);
		}
	}
}
inline void solve(int a, int b, const vector<int> &v, const vector<int> &q) {
	if (q.empty()) return;
	if (v.empty()) {
		for (int x : q) ans[x] = 0;
		return;
	}
	if (a == b) {
		vector<int> sum(q.size() + 2);
		for (int i : v) {
			int x = lower_bound(q.begin(), q.end(), i) - q.begin();
			int y = upper_bound(q.begin(), q.end(), ed[i]) - q.begin();
			sum[x] ++ , sum[y] --;
		}
		int z = 0;
		for (int i = 0; i < q.size(); i ++) {
			z += sum[i];
			ans[q[i]] = z > 0;
		}
		return ;
	}
	int mid = a + b >> 1;
	vector<int> vl, vr, vc, ql, qr, qc;
	for (int i : v) {
		if (r[i] <= mid) vl.push_back(i);
		else if (l[i] > mid) vr.push_back(i);
		else vc.push_back(i);
	}
	for (int i : q) {
		if (r[i] <= mid) ql.push_back(i);
		else if (l[i] > mid) qr.push_back(i);
		else qc.push_back(i);
	}
	work(a, b, mid, vc, vl, qc, 0);
	work(a, b, mid, vc, vr, qc, 1);
	solve(a, mid, vl, ql);
	solve(mid + 1, b, vr, qr);
}
signed main() {
	cin >> n >> m;
	n --;
	s.resize(n + 1);
	vector<int> v, q;
	for (int i = 1; i <= m; i ++) {
		cin >> tp[i];
		if (tp[i] == 2) {
			int t;
			cin >> t;
			ed[t] = i - 1;
		} else {
			cin >> l[i] >> r[i];
			r[i] --;
			if (tp[i] == 1) {
				ed[i] = m;
				v.push_back(i);
			} else {
				ans[i] = 1;
				q.push_back(i);
			}
		}
	}
	solve(1, n, v, q);
	for (int i : q) cout << (ans[i] ? 'Y' : 'N') << '\n';
	return 0;
}
```
