## 题意
给定 $n,k$，求有多少个长度为 $n$ 的序列，满足每个 $a_i\in[1,2^k-1]$，且序列 $a$ 的前缀按位或严格递增。答案对 $10^9+7$ 取模。
$$
n, k \le 3 \times 10^4
$$
## 思路

$n > k$ 必然无解。

设 $f_{i,j}$ 表示前 $i$ 个数，占了 $j$ 位的方案数，转移有 $f_{i,j} = \sum f_{i-1, j-p} \times \binom{j}{p} 2^{j-p}$，意思是枚举新选了 $p$ 位，其余 $j-p$ 位随便选。这个直接卷积的复杂度就是 $O(n^2 \log n)$ 的。

考虑优化，直接分治。设分治左右两边的区间长度分别是 $u,v$，有转移 $f_{u+v,i} = \sum \binom{i}{s} f_{u,s} f_{v,i-s} 2^{sv}$。这个直接可以卷积 $O(n \log n)$ 求出，而 $u,v,u+v$ 的取值只有 $O(\log n)$ 种，所以复杂度是 $O(n \log^2 n)$ 的。然后要用 MTT。


## 代码

```cpp
#include<bits/stdc++.h>
#define ll long long
using namespace std;
using poly = vector<int>;
const int mod = 1e9 + 7;
const int P = 998244353, Q = 1004535809, R = 469762049;
int k;
poly base;
inline int qpow(ll a, int b, int p) {
	ll res = 1;
	for (; b; b >>= 1, a = a * a % p)
		if (b & 1) res = res * a % p;
	return res;
}
template<int p>
void ntt(poly &a, bool inv) {
	int n = a.size();
	for (int i = 1, j = 0; i < n; i ++) {
		int b = n >> 1;
		while (j & b) j ^= b, b >>= 1;
		j ^= b;
		if (i < j) swap(a[i], a[j]);
	}
	for (int h = 1; h < n; h <<= 1) {
		int w = qpow(3, (p - 1) / (2 * h), p);
		for (int i = 0; i < n; i += 2 * h) {
			int t = 1;
			for (int j = i; j < i + h; j ++) {
				int x = a[j], y = (ll)a[j + h] * t % p;
				a[j] = x + y >= p ? x + y - p : x + y;
				a[j + h] = x < y ? x - y + p : x - y;
				t = (ll)t * w % p;
			}
		}
	}
	if (inv) {
		reverse(a.begin() + 1, a.end());
		int t = qpow(n, p - 2, p);
		for (int &x : a) x = (ll)x * t % p;
	}
}
template<int p>
poly conv(poly a, poly b) {
	int len = 1;
	while (len < 2 * k + 1) len <<= 1;
	for (int &x : a) x %= p;
	for (int &x : b) x %= p;
	a.resize(len), b.resize(len);
	ntt<p>(a, 0), ntt<p>(b, 0);
	for (int i = 0; i < len; i ++) a[i] = (ll)a[i] * b[i] % p;
	ntt<p>(a, 1);
	a.resize(k + 1);
	return a;
}
poly merge_poly(poly a, const poly &b, int v) {
	int w = qpow(2, v, mod), t = 1;
	for (int &x : a) {
		x = (ll)x * t % mod;
		t = (ll)t * w % mod;
	}
	poly x = conv<P>(a, b), y = conv<Q>(a, b), z = conv<R>(a, b);
	static const int ip = qpow(P, Q - 2, Q);
	static const int iq = qpow((ll)P * Q % R, R - 2, R);
	static const int pq = (ll)P * Q % mod;
	for (int i = 0; i <= k; i ++) {
		int u = (ll)(y[i] - x[i] + Q) * ip % Q;
		int s = (x[i] + (ll)P * u) % R;
		int h = (ll)(z[i] - s + R) * iq % R;
		x[i] = (x[i] + (ll)P * u + (ll)pq * h) % mod;
	}
	return x;
}
poly solve(int n) {
	if (n == 1) return base;
	int r = n >> 1;
	poly f = solve(r);
	f = merge_poly(f, f, r);
	if (n & 1) f = merge_poly(f, base, 1);
	return f;
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	ll n;
	cin >> n >> k;
	if (n > k) {
		cout << 0 << '\n';
		return 0;
	}
	base.resize(k + 1);
	int fac = 1;
	for (int i = 1; i <= k; i ++) fac = (ll)fac * i % mod;
	base[k] = qpow(fac, mod - 2, mod);
	for (int i = k; i > 1; i --)
		base[i - 1] = (ll)base[i] * i % mod;
	poly f = solve(n);
	ll ans = 0, t = 1;
	for (int i = 1; i <= k; i ++) {
		t = t * (k - i + 1) % mod;
		ans = (ans + t * f[i]) % mod;
	}
	cout << ans << '\n';
	return 0;
}
```
