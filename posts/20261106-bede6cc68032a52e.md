## 题意
有 $m$ 种卡牌，编号为 $a_1,\dots,a_m$。

每次独立、等概率抽到其中一种，重复抽到不增加已有卡牌。

不断抽卡，直到拥有的卡牌中包含 **$k$ 个连续整数编号**，求抽卡次数的期望，对 $998244353$ 取模。
$$
1 \le m \le 2×10^5,1 \le a_i \le 2m,k \le m
​$$
## 思路
太难了，这就是不直接的多项式题吗。。

定义合法是不存在连续 $k$ 张牌。设 $h_i$ 表示抽了 $i$ 张牌，且合法的方案数。距离抽到下一张不同的牌期望要 $\frac{m}{m-i}$ 步。那么有：

$$
\mathbb E[T]=\sum_{i=0}^{m-1}
\frac{h_i}{\binom mi}\cdot\frac{m}{m-i}
$$

接下来转为计数，求出所有的 $h_i$。

按编号拆成连续段，连续 $k$ 张牌的出现一定不会跨段。设 $f_{L, i}$ 表示长度为 $L$ 的连续段选了 $i$ 个数的合法方案数。考虑当前位选不选，如果选要减去不合法的方案数。有转移 $f_{L, i} \gets f_{L-1, i} + f_{L-1,i-1} - f_{L-k-1,i-k}$。因为不合法的方案一定是末尾 $k$ 个 $1$，倒数第 $k+1$ 个是 $0$。

记 $F_L(x)=\sum f_{L,i} x^i$。转移就是 
$$F_L(x) = (1+x)F_{L-1}(x)-x^k F_{L-k-1}(x)$$。
设总共有 $s$ 段，那么 
$$
g_t=[x^t]\prod_{i=1}^s F_{L_i}(x) 
$$
算 $g_t$ 直接分治 NTT 就是 $O(n \log ^2 n)$ 的。接下来只要求出 $F_L (x)$。

定义
$$
A(z)=\sum_{L\ge0}F_L(x)z^L.
$$
由递推得
$$
(1-(1+x)z+x^k z^{k+1})A(z)=1-x^kz^k
$$
所以
$$
A(z) = \frac{1-x^kz^k}{(1-(1+x)z+x^k z^{k+1})}
$$

令
$$
\sum H_L(x)z^L=\frac{1}{(1-(1+x)z+x^k z^{k+1})}
$$
那么
$$
F_L(x)=H_L(x)-x^k H_{L-k}
$$

接下来考虑求出 $H_L(x)$。
利用 $\frac{1}{1-x}=\sum_i x^i$ 展开
$$
\frac1{1-\bigl((1+x)z-x^kz^{k+1}\bigr)}
=\sum_{t\ge0}\bigl((1+x)z-x^kz^{k+1}\bigr)^t
$$
假设 $x^kz^{k+1}$ 选了 $j$ 个，那么 $(1+x)z$ 选了 $t-j$ 次，由于 $z$ 的次数是 $L$，即 $L=t-j+j(k+1)=t+jk$，所以 $t=L-jk$，
$$
H_L(x)=\sum_{j=0}^{L/(k+1)} \binom{L-jk}{j}(-x^k)^{j}(1+x)^{L-j(k+1)}
$$
令
$$
q=\left\lfloor\frac{L}{k+1}\right\rfloor,\quad
d=L-(k+1)q,
$$
再令
$$
P=-x^k,\qquad Q=(1+x)^{k+1},\qquad
c_j=\binom{L-kj}{j}.
$$ 
那么 
$$
H_L(x)=(1+x)^d\sum_{j=0}^{q}c_jP^jQ^{q-j}
$$
这个就直接分治，定义
$$
T_{l,r}=\sum_{j=l}^{r}c_jP^{j-l}Q^{r-j}
$$

$$
T_{l,r}=Q^{r-mid}T_{l,mid}+P^{mid-l+1}T_{mid+1,r}
$$
第二项直接平移系数，第一项就用二项式定理拆出组合数，做一遍 NTT 求出，求出 $H_L(x)$ 的复杂度是 $O(L \log^2 L)$，总复杂度是 $O(n \log ^2 n)$。
## 代码

```cpp
int m, k, a[N], fac[N], ifac[N];
poly_t f[N];
inline int C(int n, int m) {
	if (n < m || m < 0) return 0;
	return fac[n] * ifac[m] % mod * ifac[n - m] % mod;
}
inline poly_t binom(int n) {
	poly_t f;
	for (int i = 0; i <= n; i ++) f.push_back(C(n, i));
	return f;
}
int c[N];
inline poly_t T(int l, int r, int n) {
	if (l == r) return {c[l]};
	int mid = l + r >> 1;
	poly_t f = T(l, mid, n), g = T(mid + 1, r, n);
	poly_t ans((r - l) * (k + 1) + 1);
	int L = (mid - l + 1) * k;
	for (int i = 0; i < g.size(); i ++) {
		if ((mid - l + 1) & 1) ans[i + L] = mod - g[i];
		else ans[i + L] = g[i];
	}
	f = f * binom((r - mid) * (k + 1));
	for (int i = 0; i < f.size(); i ++) inc(ans[i], f[i]);
	return ans;
}
inline poly_t H(int n) {
	if (n < 0) return {};
	if (n <= k) return binom(n);
	int q = n / (k + 1), d = n - q * (k + 1);
	for (int j = 0; j <= q; j ++) c[j] = C(n - k * j, j);
	poly_t f = T(0, q, n);
	if (d) f = f * binom(d);
	return f;
}
inline poly_t F(int n) {
	poly_t f = H(n);
	if (n < k) return f;
	poly_t g = H(n - k);
	for (int i = 0; i < g.size(); i ++)
		dec(f[i + k], g[i]);
	return f;
}
inline poly_t solve(int l, int r) {
	if (l == r) return f[l];
	int mid = l + r >> 1;
	return solve(l, mid) * solve(mid + 1, r);
}
signed main() {
	poly :: init();
	cin >> m >> k;
	fac[0] = 1;
	for (int i = 1; i <= m; i ++) fac[i] = fac[i - 1] * i % mod;
	ifac[m] = qpow(fac[m], mod - 2);
	for (int i = m - 1; i >= 0; i --) ifac[i] = ifac[i + 1] * (i + 1) % mod;
	for (int i = 1; i <= m; i ++) cin >> a[i];
	sort (a + 1, a + m + 1);
	int cnt = 0;
	for (int l = 1, r; l <= m; l = r) {
		r = l + 1;
		while (r <= m && a[r] == a[r - 1] + 1) r ++;
		int L = r - l;
		f[++ cnt] = F(L);
	}
	poly_t h = solve(1, cnt);
	int ans = 0;
	for (int i = 0; i < m; i ++) ans = (ans + h[i] * qpow(C(m, i), mod - 2)
		                                    % mod * m % mod * qpow(m - i, mod - 2)) % mod;
	cout << ans;
	return 0;
}
```
