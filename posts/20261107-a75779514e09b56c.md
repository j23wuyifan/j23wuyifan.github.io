## 思路
设生成函数 $A(x)=\sum_{i\ge0}a_ix^i$，令
$$
Q(x)=1-\sum_{i=1}^k f_ix^i
$$
由递推式，对 $i\ge k$ 有 $[x^i](A(x)Q(x))=0$。因此 $A(x)=P(x)/Q(x)$，其中分子只需用初值计算：
$$
P(x)=\left(\sum_{i=0}^{k-1}a_ix^i\right)Q(x)\bmod x^k
$$
现在要求 $[x^n]P(x)/Q(x)$。分子分母同乘 $Q(-x)$。记
$$
U(x)=P(x)Q(-x),V(x)=Q(x)Q(-x)
$$
因为 $V(x)=V(-x)$，说明 $V(x)$ 奇数项全为 $0$，把偶次幂的指数减半，令 $Q'_i=V_{2i}$，就有 $V(x)=Q'(x^2)$。

分母只有偶次项，因此求第 $n$ 项时，分子只需保留下标与 $n$ 同奇偶的项。 令 $P'_i=U_{2i+(n\bmod 2)}$，则
$$
[x^n]\frac{P(x)}{Q(x)}
=
[x^{\lfloor n/2\rfloor}]\frac{P'(x)}{Q'(x)}.
$$
每轮令 $P\gets P'$、$Q\gets Q'$、$n\gets\lfloor n/2\rfloor$。当 $n=0$ 时，由于 $Q(0)$ 始终为 $1$，直接输出 $P(0)$。

时间复杂度 $O(k \log k \log n)$。
## 代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int mod = 998244353, G = 3, IG = 332748118;
const int N = 2e5 + 10;
inline int qpow(int a, int b) {
	int ans = 1;
	while (b) {
		if (b & 1) ans = ans * a % mod;
		a = a * a % mod, b >>= 1;
	}
	return ans;
}
inline void NTT(vector<int> &a, int G) {
	int limit = a.size();
	vector<int> r(limit);
	int L = 1;
	while ((1 << L) != limit) L ++;
	for (int i = 0; i < limit; i ++)
		r[i] = (r[i >> 1] >> 1) | ((i & 1) << (L - 1));
	for (int i = 0; i < limit; i ++)
		if (i < r[i]) swap(a[i], a[r[i]]);
	for (int mid = 1; mid < limit; mid <<= 1) {
		int wn = qpow(G, (mod - 1) / (mid << 1));
		for (int j = 0; j < limit; j += (mid << 1)) {
			int w = 1;
			for (int k = 0; k < mid; k ++, w = w * wn % mod) {
				int x = a[j + k], y = w * a[j + k + mid] % mod;
				a[j + k] = (x + y) % mod;
				a[j + k + mid] = (x - y + mod) % mod;
 			}
		}
	}
	if (G != 3) {
		int w = qpow(limit, mod - 2);
		for (int &x : a) x = x * w % mod;
	}
}
inline vector<int> mul(vector<int> a, vector<int> b) {
	int m = a.size() + b.size() - 1, n = 1;
	while (n < m) n <<= 1;
	a.resize(n), b.resize(n);
	NTT(a, G), NTT(b, G);
	for (int i = 0; i < n; i ++) a[i] = a[i] * b[i] % mod;
	NTT(a, IG);
	a.resize(m);
	return a; 
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	int n, k;
	cin >> n >> k;
	vector<int> a(k), q(k + 1);
	q[0] = 1;
	for (int i = 1, x; i <= k; i ++) {
		cin >> x;
		x = (x % mod + mod) % mod;
		q[i] = mod - x;
	}
	for (int i = 0, x; i < k; i ++) {
		cin >> x;
		a[i] = x % mod + mod;
	}
	if (n < k) {
		cout << a[n];
		return 0; 
	}
	vector<int> p = mul(a, q);
	p.resize(k);
	while (n) {
		vector<int> t = q;
		for (int i = 1; i <= k; i += 2)
			t[i] = mod - t[i];
		vector<int> u = mul(p, t), v = mul(q, t);
		for (int i = 0; i < k; i ++) p[i] = u[i * 2 + (n & 1)];
		for (int i = 0; i <= k; i ++) q[i] = v[i << 1];
		n >>= 1;
	}
	cout << (p[0] % mod + mod) % mod;
	return 0;
}
```
