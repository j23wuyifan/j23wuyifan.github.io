## 题意
定义
$$f(x,y) = \begin{cases} 2 & , x=1 \\ 2^x& , 2\le x \le 42,y = 0 \\ \prod\limits_{i=1}^{42} f(x-i,y)^i & , x \ge 43,y = 0 \\ f(x-1,y)f(x,y-1) & , x\ge 2,y \ge 1\end{cases}$$



问 $f(n,k) \bmod 998244353$。
$$
n \le 10^{18},k\le30000
$$
## 思路
将 $f(x,y)$ 变成 $\log_2 f(x,y)$。于是问题变成
一个数列 a：

$$
a_n​=n (n\le 42)
$$
$$
a_n=\sum_{i=1}^{42}i a_{n-i} (n≥43)
$$
求它 $k$ 阶前缀和的第 $n$ 项。

设原数列的生成函数为 $A(x)=\sum a_i x^i$，递推对应的分母为
$$
Q(x)=1-f_1x-f_2x^2-\cdots-f_dx^d
$$
做一次前缀和等价于生成函数除以 $1-x$，做 $k$ 次就是除以 $(1-x)^k$。相当于 $Q'(x)=Q(x)(1-x)^k$。

这个直接做个卷积，就求出 $Q'(x)$。容易发现 $Q'(x)$ 最多只有 $k+d$ 项。知道了转移系数，再做一遍常系数齐次线性递推即可。时间复杂度 $O(n \log n \log k)$。

需要注意取对数后模数是 $998244352$，要用任意模数 NTT。
## 代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
using poly = vector<int>;
const int mod = 998244353, M = mod - 1;
const int P1 = 998244353, P2 = 1004535809, P3 = 469762049;
inline int qpow(int a, int b, int p) {
	int ans = 1;
	while (b) {
		if (b & 1) ans = ans * a % p;
		a = a * a % p, b >>= 1;
	}
	return ans;
}
template<int P>
void NTT(poly &a, bool inv) {
	int n = a.size();
	static poly rt = {0, 1};
	while ((int)rt.size() < n) {
		int k = rt.size();
		int w = qpow(3, (P - 1) / (k << 1), P);
		rt.resize(k << 1);
		for (int i = k >> 1; i < k; i ++) {
			rt[i << 1] = rt[i];
			rt[i << 1 | 1] = rt[i] * w % P;
		}
	}
	for (int i = 1, j = 0; i < n; i ++) {
		int k = n >> 1;
		while (j & k) j ^= k, k >>= 1;
		j ^= k;
		if (i < j) swap(a[i], a[j]);
	}
	for (int k = 1; k < n; k <<= 1) {
		for (int i = 0; i < n; i += k << 1) {
			for (int j = 0; j < k; j ++) {
				int x = a[i + j], y = a[i + j + k] * rt[k + j] % P;
				a[i + j] = x + y >= P ? x + y - P : x + y;
				a[i + j + k] = x < y ? x - y + P : x - y;
			}
		}
	}
	if (inv) {
		reverse(a.begin() + 1, a.end());
		int w = qpow(n, P - 2, P);
		for (int &x : a) x = x * w % P;
	}
}
inline int crt(int x, int y, int z) {
	static const int i1 = qpow(P1, P2 - 2, P2);
	static const int i2 = qpow(P1 * P2 % P3, P3 - 2, P3);
	static const int p12 = P1 * P2 % M;
	int u = (y - x + P2) * i1 % P2;
	int t = (x + P1 * u) % P3;
	int v = (z - t + P3) * i2 % P3;
	return (x + P1 * u + p12 * v) % M;
}
template<int P>
void work(const poly &p, const poly &q, const poly &t, int b,
          poly &u, poly &v) {
	int d = q.size() - 1, len = 1;
	while (len <= 2 * d) len <<= 1;
	poly f(len), g(len), h(len);
	for (int i = 0; i < d; i ++) f[i] = p[i] % P;
	for (int i = 0; i <= d; i ++) {
		g[i] = q[i] % P;
		h[i] = t[i] % P;
	}
	NTT<P>(f, 0), NTT<P>(g, 0), NTT<P>(h, 0);
	for (int i = 0; i < len; i ++) {
		f[i] = f[i] * h[i] % P;
		g[i] = g[i] * h[i] % P;
	}
	NTT<P>(f, 1), NTT<P>(g, 1);
	u.resize(d), v.resize(d + 1);
	for (int i = 0; i < d; i ++) u[i] = f[i * 2 + b];
	for (int i = 0; i <= d; i ++) v[i] = g[i * 2];
}
inline int calc(poly p, poly q, int n) {
	int d = q.size() - 1;
	p.resize(d);
	poly u[3], v[3];
	while (n) {
		poly t = q;
		for (int i = 1; i <= d; i += 2)
			if (t[i]) t[i] = M - t[i];
		work<P1>(p, q, t, n & 1, u[0], v[0]);
		work<P2>(p, q, t, n & 1, u[1], v[1]);
		work<P3>(p, q, t, n & 1, u[2], v[2]);
		for (int i = 0; i < d; i ++)
			p[i] = crt(u[0][i], u[1][i], u[2][i]);
		for (int i = 0; i <= d; i ++)
			q[i] = crt(v[0][i], v[1][i], v[2][i]);
		n >>= 1;
	}
	return p[0];
}
inline poly binom(int k) {
	const int pr[] = {2, 7, 17};
	const int phi = (1LL << 22) * 6 * 16;
	int cnt[3] = {}, u = 1;
	poly a(k + 1);
	a[0] = 1;
	for (int i = 1; i <= k; i ++) {
		int x = k - i + 1, y = i;
		for (int j = 0; j < 3; j ++) {
			while (x % pr[j] == 0) x /= pr[j], cnt[j] ++;
			while (y % pr[j] == 0) y /= pr[j], cnt[j] --;
		}
		u = u * x % M * qpow(y, phi - 1, M) % M;
		int v = u;
		for (int j = 0; j < 3; j ++)
			v = v * qpow(pr[j], cnt[j], M) % M;
		a[i] = (i & 1) && v ? M - v : v;
	}
	return a;
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	int n, k;
	cin >> n >> k;
	if (n == 1) {
		cout << 2 << '\n';
		return 0;
	}
	poly a = binom(k), q(k + 43), p(42);
	for (int i = 0; i <= k; i ++) {
		q[i] = (q[i] + a[i]) % M;
		for (int j = 1; j <= 42; j ++)
			q[i + j] = (q[i + j] + M - j * a[i] % M) % M;
	}
	for (int i = 0; i < 42; i ++) {
		p[i] = i + 1;
		for (int j = 1; j <= i; j ++)
			p[i] = (p[i] + M - j * (i - j + 1)) % M;
	}
	cout << qpow(2, calc(p, q, n - 1), mod) << '\n';
	return 0;
}
```
