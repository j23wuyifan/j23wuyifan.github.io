## 题意
给定 $(s_i,a_i)_{i=1}^{n}$ 和字符串 $t$。每次询问 $(l_1,r_1,l_2,r_2)$，求
$$
\sum_{i=l_1}^{r_1}
\operatorname{occ}\!\left(s_i,\,t[l_2,r_2]\right)
\cdot \min_{l_1\le j\le i}a_j
$$
其中 $\operatorname{occ}(s,p)$ 表示 $p$ 在 $s$ 中的出现次数。强制在线。
$$
n, |S| \le 2 \times 10^5
$$
## 思路
对 $s_1\#\cdots\#s_n\#t$ 建 SAM，每加入 $s_i$ 的一个字符，就标记 $w_{dfn[u]}=i$。

倍增找到 $t[l_2,r_2]$ 对应的节点 $u$，记 $u$ 的 dfs 序的区间是 $x,y$，记 $M(l, r) = \min_{i=l}^r a_i$，查询就是 
$$
\sum_{i=x}^y [l \le w_i \le r] \cdot M(l, w_i)
$$
对 $w$ 建线段树，查询拆成 $O(\log n)$ 个线段树上的节点。对于每个线段树节点将区间内的 $w$ 排序后得到 $p_1 \dots p_k$。

对于每个查询的节点二分出 $s, e$，满足 $p_s$ 是第一个 $\ge l$ 的点，$p_e$ 是最后一个 $\le r$ 的点。查询 
$$
\sum_{i=s}^k M(l, p_i) - \sum_{i=e+1}^kM(l, p_i)
$$
这个一定是有一段前缀取 $M(l, p_s - 1)$，其余就正常取，这个也可以二分出来。预处理时间复杂度 $O(n \log n)$，一次查询 $O(\log ^2 n)$。

还没有结束。
把查询的式子稍微改一下，设 $F(v)=\sum_{p_i\ge v}M(v,p_i)$，$x$ 是 $a[l \dots r]$ 中第一个最小值的位置。答案就是 

$$
F(l)-F(x)+a_x\cdot\left|\{\,i\mid x\le p_i\le r\,\}\right|
$$
查询依旧对每个节点二分求出答案。

然后我们稍微观察一下二分的东西，相当于是查询 $O(\log n)$ 个序列中的后继，这不是分散层叠吗？上个分散层叠的板子即可做到 $O((n+q) \log n)$。
## 代码

```cpp
#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 2e5 + 10;
int n, q, st[19][N], lg[N], pos[N], w[N], m;
const int M = 6 * N;
struct SAM {
	int fa, ch[27], len;
} a[M];
int tot = 1, lst = 1, dep, tag[M], fa[21][M];
int in[M], out[M], stk[M], top;
vector<int> E[M];
inline void insert(int v, int id) {
	int u = ++ tot, p = lst;
	a[u].len = a[lst].len + 1;
	while (p && !a[p].ch[v]) a[p].ch[v] = u, p = a[p].fa;
	if (!p) a[u].fa = 1;
	else {
		int q = a[p].ch[v];
		if (a[p].len + 1 == a[q].len) a[u].fa = q;
		else {
			int nw = ++ tot;
			a[nw] = a[q];
			a[nw].len = a[p].len + 1, a[u].fa = a[q].fa = nw;
			while (p && a[p].ch[v] == q) a[p].ch[v] = nw, p = a[p].fa;
		}
	}
	tag[u] = id, lst = u;
}
inline void init() {
	for (int i = 2; i <= tot; i ++) {
		fa[0][i] = a[i].fa;
		E[a[i].fa].emplace_back(i);
	}
	stk[++ top] = 1;
	while (top) {
		int u = stk[top --];
		if (u < 0) {
			out[-u] = m;
			continue;
		}
		in[u] = m + 1;
		if (tag[u]) w[++ m] = tag[u];
		stk[++ top] = -u;
		for (int v : E[u]) stk[++ top] = v;
	}
	while ((1 << (dep + 1)) <= a[lst].len) dep ++;
	for (int j = 1; j <= dep; j ++)
		for (int i = 1; i <= tot; i ++)
			fa[j][i] = fa[j - 1][fa[j - 1][i]];
}
int v[N], pre[N], nxt[N];
inline int chk(int x, int y) {
	return v[x] <= v[y] ? x : y;
}
inline int qry(int l, int r) {
	int k = lg[r - l + 1];
	return chk(st[k][l], st[k][r - (1 << k) + 1]);
}
const int block = 24;
struct info {
	int key, j, h;
};
struct point {
	int key, seg, left, right;
};
struct node {
	int *p, sz;
	ll *g;
	vector<info> seg;
	vector<point> tab;
};
node tr[32768];
int val[15][N], b[N];
ll sum[15][N];
info tmp[N];
inline void split(node &t) {
	auto &p = t.p;
	int k = t.sz, top = 0;
	int num = 0;
	t.seg.push_back({p[k - 1] + 1, k, k});
	for (int j = k - 1; j >= 0; j --) {
		int lo = j ? p[j - 1] + 1 : 1;
		b[j] = j ? v[qry(p[j - 1], p[j])] : 0;
		int z = top;
		while (z && b[stk[z - 1]] >= b[j]) z --;
		int h = z ? stk[z - 1] : k;
		t.g[j] = 1LL * (h - j) * b[j] + (h < k ? t.g[h] : 0);
		if (lo <= p[j]) {
			num = 1;
			tmp[1] = {lo, j, k};
			for (int i = max(0, z - 1); i < top; i ++) {
				h = stk[i];
				int u = qry(p[j], p[h]);
				if (u == p[j]) continue;
				int key = max(lo, pre[u] + 1);
				if (tmp[num].key == key) tmp[num].h = h;
				else tmp[++ num] = {key, j, h};
			}
			while (num) t.seg.push_back(tmp[num --]);
		}
		top = z, stk[top ++] = j;
	}
	reverse(t.seg.begin(), t.seg.end());
}
#define lc id << 1
#define rc id << 1 | 1
inline void build(int id, int l, int r, int d) {
	auto &t = tr[id];
	t.sz = r - l + 1, t.p = val[d] + l, t.g = sum[d] + l;
	if (r - l + 1 <= block) {
		copy(w + l, w + r + 1, t.p);
		sort(t.p, t.p + t.sz);
		return;
	}
	int mid = (l + r) >> 1;
	build(lc, l, mid, d + 1), build(rc, mid + 1, r, d + 1);
	auto &u = tr[lc], &v = tr[rc];
	merge(u.p, u.p + u.sz, v.p, v.p + v.sz, t.p);
	split(t);
	int i = 0, j = 1, h = 1, x = 0, y = 0;
	while (i < t.seg.size() || j < u.tab.size() || h < v.tab.size()) {
		int key = n + 2;
		if (i < t.seg.size()) key = min(key, t.seg[i].key);
		if (j < u.tab.size()) key = min(key, u.tab[j].key);
		if (h < v.tab.size()) key = min(key, v.tab[h].key);
		if (i < t.seg.size() && t.seg[i].key == key) i ++;
		if (j < u.tab.size() && u.tab[j].key == key) j += 2;
		if (h < v.tab.size() && v.tab[h].key == key) h += 2;
		while (x < u.tab.size() && u.tab[x].key <= key) x ++;
		while (y < v.tab.size() && v.tab[y].key <= key) y ++;
		t.tab.push_back({key, i - 1, x, y});
	}
}
int key[3], ql, qr, c;
struct ranks {
	int p[3];
};
inline ranks down(int id, const ranks &s, int d) {
	ranks t;
	auto &v = tr[id * 2 + d].tab;
	for (int i = 0; i < 3; i ++) {
		auto e = tr[id].tab[s.p[i] - 1];
		int j = d ? e.right : e.left;
		if (j < v.size() && v[j].key <= key[i]) j ++;
		t.p[i] = j;
	}
	return t;
}
inline ll suf(const node &t, info s, int x) {
	if (s.j == t.sz) return 0;
	return 1LL * (s.h - s.j) * v[qry(x, t.p[s.j])] + (s.h < t.sz ? t.g[s.h] : 0);
}
inline ll query(int id, int l, int r, const ranks &s) {
	if (r - l + 1 <= block) {
		ll res = 0;
		for (int i = max(l, ql); i <= min(r, qr); i ++)
			if (key[0] <= w[i] && w[i] < key[1]) res += v[qry(key[0], w[i])];
		return res;
	}
	auto &t = tr[id];
	if (ql <= l && r <= qr) {
		info f = t.seg[t.tab[s.p[0] - 1].seg], g = t.seg[t.tab[s.p[1] - 1].seg];
		info h = t.seg[t.tab[s.p[2] - 1].seg];
		return suf(t, f, key[0]) - suf(t, h, key[2]) - 1LL * c * (h.j - g.j);
	}
	int mid = (l + r) >> 1;
	ll res = 0;
	if (ql <= mid) res += query(lc, l, mid, down(id, s, 0));
	if (qr > mid) res += query(rc, mid + 1, r, down(id, s, 1));
	return res;
}
inline ll ask(int u, int l, int r) {
	ql = in[u], qr = out[u];
	if (ql > qr || l > r) return 0;
	int x = qry(l, r);
	c = v[x], key[0] = l, key[1] = r + 1, key[2] = nxt[x];
	ranks s{};
	for (int i = 0; i < 3; i ++) {
		int low = 0, high = tr[1].tab.size();
		while (low < high) {
			int mid = (low + high) >> 1;
			if (tr[1].tab[mid].key <= key[i]) low = mid + 1;
			else high = mid;
		}
		s.p[i] = low;
	}
	return query(1, 1, m, s);
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	int sid, o, fl, fr;
	cin >> sid >> n >> q >> o >> fl >> fr;
	string s;
	for (int i = 1; i <= n; i ++) {
		cin >> s;
		for (char c : s) insert(c - 'a', i);
		insert(26, 0);
	}
	cin >> s;
	int t = s.size();
	for (int i = 1; i <= t; i ++) insert(s[i - 1] - 'a', 0), pos[i] = lst;
	init();
	for (int i = 1; i <= n; i ++) cin >> v[i], st[0][i] = i;
	for (int i = 2; i <= n; i ++) lg[i] = lg[i >> 1] + 1;
	for (int j = 1; (1 << j) <= n; j ++)
		for (int i = 1; i + (1 << j) - 1 <= n; i ++)
			st[j][i] = chk(st[j - 1][i], st[j - 1][i + (1 << (j - 1))]);
	top = 0;
	for (int i = 1; i <= n; i ++) {
		while (top && v[stk[top]] > v[i]) nxt[stk[top --]] = i;
		pre[i] = top ? stk[top] : 0;
		stk[++ top] = i;
	}
	while (top) nxt[stk[top --]] = n + 1;
	build(1, 1, m, 0);
	ll ans = 0;
	while (q --) {
		int l, r, x, y;
		cin >> l >> r >> x >> y;
		ll d = o * ans;
		l = (l + d - 1) % n + 1, r = (r + d - 1) % n + 1;
		x = (x + d - 1) % t + 1, y = (y + d - 1) % t + 1;
		if (l > r) swap(l, r);
		if (x > y) swap(x, y);
		if (fl != -1) l = fl;
		if (fr != -1) r = fr;
		int u = pos[y];
		for (int j = dep; j >= 0; j --) {
			int v = fa[j][u];
			if (a[v].len >= y - x + 1) u = v;
		}
		ans = ask(u, l, r);
		cout << ans << '\n';
	}
	return 0;
}
```
