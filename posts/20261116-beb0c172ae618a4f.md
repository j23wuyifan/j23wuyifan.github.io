## 题意
给定 $n$ 个有编号的点，每个点 $u$ 有点权 $a_u$。考虑这些点组成的**所有有根树**，根可以任选。

在一棵树上，初始只知道点 $s$。每次花费 $1$，可以用以下任一方式得知点 $v$：

- **直接操作**：选一个已知点 $u$，满足 $a_u>a_v$，且 $v$ 不在 $u$ 的子树内。
- **支点操作**：选两个已知点 $u,w$，满足 $w$ 是 $u$ 的祖先，且 $a_u>a_w>a_v$。此时 $v$ 的位置没有限制。

记 $f(s,v)$ 为从只知道 $s$ 到得知 $v$ 的最少操作次数，无法得知则记为 $0$。

求**所有有根树、所有有序点对 $s\ne v$** 的 $f(s,v)$ 之和，对 $998244353$ 取模。
$$
n \le 2 \times 10^5,a_i \le 10^9
$$
## 思路
容易发现 $f(u,v) = 0/1/2$。因为若 $a_u > a_v$, 有两种情况：
- $v$ 不在 $u$ 子树内，$f(u,v)=1$。
- $v$ 在 $u$ 子树内，若存在 $a_v < a_w < a_u$ 且 $w$ 不在 $u$ 子树内，那么 $w$ 必定可以通过一次操作到 $v$。

设有 $t$ 个 $a_v < a_w < a_u$ 的 $w$。记 $d_i$ 为 $u$ 子树内钦定 $i$ 个 $w$ 和 $v$ 必选的方案数。$f(u,v)$ 的答案就是
$$
(n ^ {n - 1} - d_0) + 2(d_0 - d_t)
$$
考虑如何求出 $d_t$。
$$
d_t = \sum_{i=0}^{n-t-2} \binom{n-t-2}{i} (n-i)^{n-i-2} i^i
$$
后面两项分别是 $u$ 子树内的方案和 $u$ 子树外选 $u$ 的父亲和树根的方案数。这个做一遍卷积即可求出。

考虑 $d_t$ 对多少对 $u,v$ 有贡献，排序后设最后一个 $a_v$ 的位置是 $x$，第一个 $a_u$ 的位置是 $y$，$t = y - x - 1$，这个也是做一遍卷积即可。

时间复杂度 $O(n \log n)$。
## 代码

```cpp
#include<bits/stdc++.h>
#define poly vector<int>
#define int long long
using namespace std;
const int N = (1 << 19) + 10, mod = 998244353;
const int G = 3, IG = (mod + 1) / 3;
int n, a[N], rev[N];
inline int qpow(int a, int b) {
	int ans = 1;
	while (b) {
		if (b & 1) ans = ans * a % mod;
		a = a * a % mod, b >>= 1;
	}
	return ans;
}
inline void ntt(poly &a, int G) {
	int L = 1, n = a.size();
	while ((1 << L) != n) L ++;
	for (int i = 0; i < n; i ++) rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (L - 1));
	for (int i = 0; i < n; i ++) if (i < rev[i]) swap(a[i], a[rev[i]]);
	for (int mid = 1; mid < n; mid <<= 1) {
		int wn = qpow(G, (mod - 1) / (mid << 1));
		for (int j = 0; j < n; j += mid << 1) {
			for (int k = 0, w = 1; k < mid; k ++, w = w * wn % mod) {
				int x = a[j + k], y = w * a[j + k + mid] % mod;
				a[j + k] = (x + y) % mod;
				a[j + k + mid] = (x - y + mod) % mod; 
			}
		}
	}
	if (G != 3) {
		const int inv = qpow(n, mod - 2);
		for (int i = 0; i < n; i ++) a[i] = a[i] * inv % mod;
	} 
}
inline poly mul(poly &a, poly &b) {
	int n = a.size() + b.size() - 1, L = 1;
	while ((1 << L) < n) L ++;
	a.resize(1 << L), b.resize(1 << L);
	ntt(a, G), ntt(b, G);
	for (int i = 0; i < (1 << L); i ++) a[i] = a[i] * b[i] % mod;
	ntt(a, IG);
	a.resize(n);
	return a;
}
int fac[N], ifac[N], d[N];
signed main() {
	cin >> n;
	for (int i = 1; i <= n; i ++) cin >> a[i];
	sort (a + 1, a + n + 1);
	fac[0] = 1;
	for (int i = 1; i <= n; i ++) fac[i] = fac[i - 1] * i % mod;
	ifac[n] = qpow(fac[n], mod - 2);
	for (int i = n - 1; i >= 0; i --) ifac[i] = ifac[i + 1] * (i + 1) % mod;
	poly f(n - 1, 0), g(n - 1, 0);
	for (int i = 0; i <= n - 2; i ++) f[i] = ifac[i] * qpow(i, i) % mod * qpow(n - i, n - i - 2) % mod;
	for (int i = 0; i <= n - 2; i ++) g[i] = ifac[i];
	f = mul(f, g);
	for (int i = 0; i <= n - 2; i ++) d[i] = fac[i] * f[i] % mod;
	reverse(d, d + n - 2 + 1);
	poly p(n + 1, 0), q(n + 1, 0);
	int ans = 0;
	for (int l = 1, r; l <= n; l = r + 1) {
		r = l;
		while (a[r + 1] == a[l]) r ++;
		q[n - r] = p[l] = r - l + 1;
		ans += (r - l + 1) * (n - r);
	} 
	p = mul(p, q);
	ans = ans % mod * (qpow(n, n - 1) + d[0]) % mod;
	for (int i = 0; i <= n - 2; i ++) {
		int v = p[n + 1 + i];
		ans = (ans - 2 * v * d[i]) % mod;
	}
	cout << (ans + mod) % mod;
	return 0;
}
```
