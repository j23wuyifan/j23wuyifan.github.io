## 题意
给定 $n$ 个城市，第 $i$ 个城市可以在 $[l_i,r_i]$ 内任选一个整数日期访问，也可以跳过。

要求选中的城市编号严格递增，访问日期也严格递增。求最多能访问多少个城市。
## 思路
简单题。

题意相当于是把每个位置选一个 $p_i \in [l_i,r_i]$，求 LIS。

设 $f_i$ 表示长度为 $i$ 的最小结尾。考虑新的一个 $[l,r]$ 对 $f$ 的贡献。有转移
$$
f_i = \min(f_i,\max(f_{i-1}+1,l))(f_{i-1} < r)
$$
这相当于是插入一个 $l$，把 $f_j \in [l,r)$ 都 $+1$，再删掉第一个 $\ge r$ 的 $f_j$。

用平衡树维护即可。
## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 3e5 + 10;
int n, rt, val[N], lc[N], rc[N], tot, tag[N], sz[N], rnd[N];
mt19937 rd(78923432);
inline void apply(int id, int x) {
	val[id] += x, tag[id] += x;
}
inline void pushdown(int id) {
	if (lc[id]) apply(lc[id], tag[id]);
	if (rc[id]) apply(rc[id], tag[id]);
	tag[id] = 0;
}
inline void pushup(int id) {
	sz[id] = sz[lc[id]] + sz[rc[id]] + 1;
}
inline void split(int u, int k, int &l, int &r) {
	if (!u) {
		l = r = 0;
		return;
	}
	pushdown(u);
	if (val[u] < k) {
		l = u;
		split(rc[u], k, rc[u], r);
		pushup(u);
	}
	else {
		r = u;
		split(lc[u], k, l, lc[u]);
		pushup(u);
	}
}
inline int merge(int l, int r) {
	if (!l || !r) return l | r;
	pushdown(l), pushdown(r);
	if (rnd[l] < rnd[r]) {
		rc[l] = merge(rc[l], r);
		pushup(l);
		return l;
	}
	else {
		lc[r] = merge(l, lc[r]);
		pushup(r);
		return r;
	}
}
inline int del(int u) {
	if (!u) return 0;
	pushdown(u);
	if (!lc[u]) return rc[u];
	lc[u] = del(lc[u]);
	pushup(u);
	return u;
}
inline int newnode(int x) {
	val[++ tot] = x, rnd[tot] = rd(), sz[tot] = 1;
	return tot;
}
signed main() {
	cin >> n; 
	for (int i = 1, l, r; i <= n; i ++) {
		cin >> l >> r;
		int a, b;
		split(rt, l, rt, a);
		split(a, r, a, b);
		apply(a, 1);
		b = del(b);
		rt = merge(merge(rt, newnode(l)), merge(a, b));
	}
	cout << sz[rt];
	return 0;
}
```
