## 题意
给定一个排列 $p$，把每个位置看作一个点，每个逆序对连一条无向边。对每个点，求它到同一连通块内所有点的最短距离之和。
$$
n \le 2 \times 10^5
$$
## 思路
挺厉害的题。

先求出所有的 $j<i$ 的贡献，坐标翻转再做一遍就可以得出答案。

若 $\max(p_1 \dots p_r)=r$，在 $r$ 这里新分一段，最后分成若干个连通块。

从 $i$ 出发，走 $k$ 步能到的最小下标是 $x$，最小的 $p_i$ 是 $y$。

对于左侧点 $j<i$，有
$$
\operatorname{dist}(i,j)>k+1
\iff j<x\ \text{且}\ p_j<y
$$
因为不可达，所以连通块都在 $i$ 点的右上方和左下方。但是右上和左下不联通，所以联通都在右上方。

记
$$
L_y=\min_{p_j\ge y}j,\qquad R_x=\min_{j\ge x}p_j
$$

$(x,y)$ 再走一步到达 $(L_y,R_x)$。

$F(x,y)$ 表示当前走到 $(x,y)$，要到达所有点还要走的边数和。
有转移
$$
F(x,y)=F(L_y,R_x)+C(x,y)
$$
其中 $C(x,y)=\sum [j<x \land p_j<y]$。

稍微分析一下容易发现 $F(x,y)$ 只有 $O(n \sqrt{n})$ 种取值。设 $B\approx\sqrt n$。
- 若 $y-R_x>B$，从同一起点出发，这种转移最多 $n/B$ 次。
- 若 $y-R_x\le B$，固定 $x$ 后，$y\in[R_x,R_x+B]$，只有 $B+1$ 种状态。

$C(x,y)$ 是一个二维数点，$O(n)$ 个加法，$O(n\sqrt{n})$ 次查询。分块即可做到 $O(\sqrt{n})-O(1)$。

总时间复杂度为 $O(n\sqrt{n})$
## 代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 2e5 + 10, B = 450;
int n, p[N], pos[N], l[N], r[N], ans[N], a[N], tag[N];
struct node {
    signed y, id;
    int f;
};
vector<node> q[N];
inline int ins(int x, int y) {
    if (!q[x].size() || q[x].back().y != y)
        q[x].push_back({y, 0, 0});
    return q[x].size() - 1;
}
inline void add(int x, int R) {
    int k = x / B;
    for (int i = x; i <= min(R, (k + 1) * B - 1); i ++) a[i] ++;
    for (int i = k + 1; i <= R / B; i ++) tag[i] ++;
}
inline int qry(int x) {
    return a[x] + tag[x / B];
}
inline void work(int L, int R) {
    if (L == R) return ;
    l[R] = pos[R], r[R] = p[R];
    for (int i = R - 1; i >= L; i --) {
        l[i] = min(l[i + 1], pos[i]);
        r[i] = min(r[i + 1], p[i]);
    }
    for (int x = R; x >= L; x --) {
        ins(x, p[x]);
        int fl = 0;
        for (auto &u : q[x]) {
            int t = l[u.y];
            if (t == x) fl = 1;
            else u.id = ins(t, r[x]);
        }
        if (!fl) continue;
        int k = ins(x, r[x]), t = l[r[x]];
        if (t < x) q[x][k].id = ins(t, r[x]);
        for (auto &u : q[x])
            if (l[u.y] == x) u.id = k;
    }
    for (int x = L; x <= R; x ++) {
        int m = q[x].size();
        for (int k = m - 1; k >= 0; k --) {
            auto &u = q[x][k];
            int y = u.y, t = l[y];
            u.f = qry(y - 1);
            if (t != x || y != r[x]) u.f += q[t][u.id].f;
            if (y == p[x]) ans[x] += x - L + u.f;
        }
        add(p[x], R);
    }
    for (int i = L; i <= R; i ++) {
        a[i] = 0;
        vector<node>().swap(q[i]);
    }
    fill(tag + L / B, tag + R / B + 1, 0);
}
inline void calc() {
    for (int i = 1; i <= n; i ++) pos[p[i]] = i;
    int L = 1, mx = 0;
    for (int i = 1; i <= n; i ++) {
        mx = max(mx, p[i]);
        if (mx == i) work(L, i), L = i + 1;
    }
}
inline void solve() {
    cin >> n;
    for (int i = 1; i <= n; i ++) cin >> p[i], ans[i] = 0;
    calc();
    reverse(p + 1, p + n + 1);
    reverse(ans + 1, ans + n + 1);
    for (int i = 1; i <= n; i ++) p[i] = n - p[i] + 1;
    calc();
    for (int i = n; i >= 1; i --) 
        cout << ans[i] << ' ';
    cout << '\n';
}
signed main() {
    ios :: sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    int T = 1;
    // cin >> T;
    while (T --) solve();
    return 0;
}
```
