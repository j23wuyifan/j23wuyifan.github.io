## 题意
给树上 $m$ 条路径，统计有多少无序点对同时落在至少一条给定路径上。$n,m\le10^5$。
## 思路
先给个我自己的做法：

考虑如果树是一条链怎么做，相当于是矩形加，查询矩形内 $> 0$ 的数有多少个。扫描线+线段树容易做到 $O(n \log n)$。

再考虑一般的情况。考虑树剖，一条路径上有 $O(\log n)$ 个区间。直接两两枚举贡献到矩形上即可。

有 $O(n \log^2 n)$ 个区间，总复杂度 $O(n \log^3 n)$。不卡常可以轻松通过。
```cpp
#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 1e5 + 10;
int n, m, tot = 1, hd[N], to[N << 1], nxt[N << 1];
int fa[N], dep[N], sz[N], son[N], top[N], dfn[N], tim;
int cnt[N << 2], len[N << 2];
vector<pair<int, int> > seg[N];
vector<int> ev[N];
inline void add(int u, int v) {
	to[++ tot] = v, nxt[tot] = hd[u], hd[u] = tot;
}
inline void dfs1(int u, int p) {
	fa[u] = p, dep[u] = dep[p] + 1, sz[u] = 1;
	for (int e = hd[u]; e; e = nxt[e]) {
		int v = to[e];
		if (v == p) continue;
		dfs1(v, u);
		sz[u] += sz[v];
		if (sz[v] > sz[son[u]]) son[u] = v;
	}
}
inline void dfs2(int u, int t) {
	top[u] = t, dfn[u] = ++ tim;
	if (son[u]) dfs2(son[u], t);
	for (int e = hd[u]; e; e = nxt[e]) {
		int v = to[e];
		if (v != fa[u] && v != son[u]) dfs2(v, v);
	}
}
inline void split(int u, int v, vector<pair<int, int>> &a) {
	while (top[u] != top[v]) {
		if (dep[top[u]] < dep[top[v]]) swap(u, v);
		a.emplace_back(dfn[top[u]], dfn[u]);
		u = fa[top[u]];
	}
	if (dfn[u] > dfn[v]) swap(u, v);
	a.emplace_back(dfn[u], dfn[v]);
	sort(a.begin(), a.end());
	int k = 0;
	for (auto [l, r] : a) {
		if (k && l == a[k - 1].second + 1) a[k - 1].second = r;
		else a[k ++] = {l, r};
	}
	a.resize(k);
}
inline void pushup(int id, int l, int r) {
	if (cnt[id]) len[id] = r - l + 1;
	else if (l == r) len[id] = 0;
	else len[id] = len[id << 1] + len[id << 1 | 1];
}
inline void upd(int id, int l, int r, int x, int y, int v) {
	if (x <= l && r <= y) {
		cnt[id] += v;
	} else {
		int mid = (l + r) >> 1;
		if (x <= mid) upd(id << 1, l, mid, x, y, v);
		if (y > mid) upd(id << 1 | 1, mid + 1, r, x, y, v);
	}
	pushup(id, l, r);
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> m;
	for (int i = 1, u, v; i < n; i ++) {
		cin >> u >> v;
		add(u, v), add(v, u);
	}
	dfs1(1, 0), dfs2(1, 1);
	for (int i = 1, u, v; i <= m; i ++) {
		cin >> u >> v;
		if (u == v) continue;
		split(u, v, seg[i]);
		for (auto [l, r] : seg[i]) {
			ev[l].push_back(i);
			if (r < n) ev[r + 1].push_back(-i);
		}
	}
	ll ans = 0;
	for (int x = 1; x <= n; x ++) {
		for (int p : ev[x]) {
			int v = p > 0 ? 1 : -1;
			for (auto [l, r] : seg[abs(p)]) upd(1, 1, n, l, r, v);
		}
		ans += len[1] - (len[1] > 0);
	}
	cout << ans / 2 << '\n';
	return 0;
}
```

接下来讲一下正解。

考虑固定一个端点是 $u$，可以选择的方案数。

求出 $s_u$ 表示经过 $u$ 的路径的端点的集合，$u$ 的贡献就是使 $s_u$ 联通的最小联通子树的边数。将 $s_u$ 按照 dfn 序排序后是 $a_1 \dots a_k$，那么贡献也就是 
$$
\frac{1}{2}\sum_{i=1}^k dist(a_i,a_{i+1})
$$
这个可以用线段树维护。

剩下就是求出 $s_u$ 了，对于路径 $u,v$，在 $u,v$ 上 +1，在 $lca(u,v),fa_{lca(u,v)}$ 处 -1，然后做线段树合并即可。

若用欧拉序求 lca 可以做到 $O(n \log n)$。
## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int n, m, tot = 1, hd[N], to[N << 1], nxt[N << 1];
int fa[N], dep[N], dfn[N], seq[N], fir[N], tim, ec;
int st[18][N << 1], lg[N << 1], Id;
long long ans;
vector<int> ev[N];
struct Node {
	int lc, rc, a, b, sum, cnt;
};
deque<Node> tr(1);
inline void add(int u, int v) {
	to[++ tot] = v, nxt[tot] = hd[u], hd[u] = tot;
}
inline void dfs1(int u, int p) {
	fa[u] = p, dep[u] = dep[p] + 1;
	dfn[u] = ++ tim, seq[tim] = u;
	fir[u] = ++ ec, st[0][ec] = u;
	for (int e = hd[u]; e; e = nxt[e]) {
		int v = to[e];
		if (v == p) continue;
		dfs1(v, u);
		st[0][++ ec] = u;
	}
}
inline int get(int u, int v) {
	return dep[u] < dep[v] ? u : v;
}
inline int lca(int u, int v) {
	int l = fir[u], r = fir[v];
	if (l > r) swap(l, r);
	int k = lg[r - l + 1];
	return get(st[k][l], st[k][r - (1 << k) + 1]);
}
inline int dis(int u, int v) {
	return dep[u] + dep[v] - 2 * dep[lca(u, v)];
}
inline int alloc() {
	if (!Id) {
		tr.push_back({});
		return (int)tr.size() - 1;
	}
	int p = Id;
	Id = tr[p].lc, tr[p] = {};
	return p;
}
inline int del(int p) {
	tr[p].lc = Id, Id = p;
	return 0;
}
inline void pushup(int p) {
	const Node &x = tr[tr[p].lc], &y = tr[tr[p].rc];
	tr[p].a = x.a ? x.a : y.a;
	tr[p].b = y.b ? y.b : x.b;
	tr[p].sum = x.sum + y.sum;
	if (x.b && y.a) tr[p].sum += dis(x.b, y.a);
}
inline int leaf(int p, int l) {
	if (!tr[p].cnt) return del(p);
	tr[p].a = tr[p].b = tr[p].cnt > 0 ? seq[l] : 0;
	return p;
}
inline int upd(int p, int l, int r, int x, int v) {
	if (!p) p = alloc();
	if (l == r) {
		tr[p].cnt += v;
		return leaf(p, l);
	}
	int mid = (l + r) >> 1;
	if (x <= mid) tr[p].lc = upd(tr[p].lc, l, mid, x, v);
	else tr[p].rc = upd(tr[p].rc, mid + 1, r, x, v);
	if (!tr[p].lc && !tr[p].rc) return del(p);
	pushup(p);
	return p;
}
inline int merge(int p, int q, int l, int r) {
	if (!p || !q) return p | q;
	if (l == r) {
		tr[p].cnt += tr[q].cnt;
		del(q);
		return leaf(p, l);
	}
	int mid = (l + r) >> 1;
	tr[p].lc = merge(tr[p].lc, tr[q].lc, l, mid);
	tr[p].rc = merge(tr[p].rc, tr[q].rc, mid + 1, r);
	del(q);
	if (!tr[p].lc && !tr[p].rc) return del(p);
	pushup(p);
	return p;
}
inline int dfs2(int u) {
	int p = 0;
	for (int e = hd[u]; e; e = nxt[e]) {
		int v = to[e];
		if (v != fa[u]) p = merge(p, dfs2(v), 1, n);
	}
	for (int x : ev[u]) p = upd(p, 1, n, abs(x), x > 0 ? 1 : -1);
	if (tr[p].a) ans += tr[p].sum + dis(tr[p].a, tr[p].b);
	return p;
}
inline void mark(int u, int x, int y) {
	if (!u) return;
	ev[u].push_back(x), ev[u].push_back(y);
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> m;
	for (int i = 1, u, v; i < n; i ++) {
		cin >> u >> v;
		add(u, v), add(v, u);
	}
	dfs1(1, 0);
	for (int i = 2; i <= ec; i ++) lg[i] = lg[i >> 1] + 1;
	for (int k = 1; k <= lg[ec]; k ++) {
		for (int i = 1; i + (1 << k) - 1 <= ec; i ++) {
			st[k][i] = get(st[k - 1][i], st[k - 1][i + (1 << (k - 1))]);
		}
	}
	for (int i = 1, s, t; i <= m; i ++) {
		cin >> s >> t;
		if (s == t) continue;
		int w = lca(s, t), x = dfn[s], y = dfn[t];
		mark(s, x, y), mark(t, x, y);
		mark(w, -x, -y), mark(fa[w], -x, -y);
	}
	dfs2(1);
	cout << ans / 4 << '\n';
	return 0;
}
```
