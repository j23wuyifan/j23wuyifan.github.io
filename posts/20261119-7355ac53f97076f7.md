## 题意
带权树，每次给 $x,y$，选 $y$ 条路径，使它们的并连通、经过 $x$，覆盖边权和最大。强制在线。

$n \le 10^5$
## 思路
选 $y$ 条经过 $x$ 的路径等价于选一个以 $x$ 为根 $2y$ 个叶子的联通子树。构造就是将叶子按照 dfn 序排序，隔 $k$ 个的连边即可。

考虑如何对一个 $x$ 求答案。就是长剖后取最长的 $2k$ 条长链即可。

如果没有强制在线，考虑换根，设 $u$ 是 $v$ 的父亲，边权是 $w$。设 $u$ 方向的最长链长度是 $a$，$v$ 方向的最长链长度是 $b$。

那么新的长链集合中有 $a \gets a+w,b \gets b-w$。其余不变。

设经过 $v$ 的次长链是 $t$，全体第 $2k$ 长链是 $b$。若 $b>t$，就要用 $t$ 替换 $b$，因为 $v$ 至少要有两侧的点，这样联通子树才经过 $v$。

强制在线就可持久化线段树即可。

时空复杂度 $O(n \log n)$。
## 代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e5 + 10;
int n, q, m, tot, ans, rt[N], fa[N], son[N], d[N], w[N], a[N], b[N];
vector<pair<int, int> > e[N];
vector<int> val;
struct node {
	signed l, r, cnt, sum;
} tr[N * 100];
inline void dfs1(int u, int f) {
	fa[u] = f;
	for (auto [v, c] : e[u]) {
		if (v == f) continue;
		w[v] = c;
		dfs1(v, u);
		if (d[v] + c > d[u]) {
			d[u] = d[v] + c;
			son[u] = v;
		}
	}
}
inline void dfs2(int u, int up) {
	a[u] = up;
	for (auto [v, c] : e[u]) {
		if (v == fa[u]) continue;
		int t = d[v] + c;
		if (t > a[u]) b[u] = a[u], a[u] = t;
		else b[u] = max(b[u], t);
	}
	val.push_back(d[u]);
	val.push_back(a[u]);
	val.push_back(b[u]);
	for (auto [v, c] : e[u]) {
		if (v == fa[u]) continue;
		int t = d[v] + c;
		int h = (t == a[u] ? b[u] : a[u]);
		val.push_back(t);
		val.push_back(h + c);
		dfs2(v, h + c);
	}
}
inline int upd(int p, int l, int r, int x, int k) {
	int id = ++ tot;
	tr[id] = tr[p];
	tr[id].cnt += k;
	tr[id].sum += val[x - 1] * k;
	if (l == r) return id;
	int mid = (l + r) >> 1;
	if (x <= mid) tr[id].l = upd(tr[p].l, l, mid, x, k);
	else tr[id].r = upd(tr[p].r, mid + 1, r, x, k);
	return id;
}
inline int add(int p, int v, int k) {
	if (!v) return p;
	int x = lower_bound(val.begin(), val.end(), v) - val.begin() + 1;
	return upd(p, 1, m, x, k);
}
inline void dfs3(int u) {
	for (auto [v, c] : e[u]) {
		if (v == fa[u]) continue;
		int t = d[v] + c;
		int h = (t == a[u] ? b[u] : a[u]);
		int p = rt[u];
		if (h != d[v]) {
			p = add(p, h, -1);
			p = add(p, t, -1);
			p = add(p, h + c, 1);
			p = add(p, d[v], 1);
		}
		rt[v] = p;
		dfs3(v);
	}
}
inline int qry(int p, int l, int r, int k, int &t) {
	if (k > tr[p].cnt) {
		t = 0;
		return tr[p].sum;
	}
	if (l == r) {
		t = val[l - 1];
		return t * k;
	}
	int mid = (l + r) >> 1, rc = tr[p].r;
	if (k <= tr[rc].cnt) return qry(rc, mid + 1, r, k, t);
	return tr[rc].sum + qry(tr[p].l, l, mid, k - tr[rc].cnt, t);
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	cin >> n >> q;
	for (int i = 1, u, v, c; i < n; i ++) {
		cin >> u >> v >> c;
		e[u].push_back({v, c});
		e[v].push_back({u, c});
	}
	dfs1(1, 0);
	dfs2(1, 0);
	sort(val.begin(), val.end());
	val.erase(unique(val.begin(), val.end()), val.end());
	m = val.size();
	rt[1] = add(0, d[1], 1);
	for (int u = 2; u <= n; u ++)
		if (son[fa[u]] != u) rt[1] = add(rt[1], d[u] + w[u], 1);
	dfs3(1);
	while (q --) {
		int x, y, t;
		cin >> x >> y;
		x = (x + ans - 1) % n + 1;
		y = (y + ans - 1) % n + 1;
		ans = qry(rt[x], 1, m, 2 * y, t);
		ans -= max(0LL, t - b[x]);
		cout << ans << '\n';
	}
	return 0;
}
```
