## 题意
给定一棵 $n$ 个点的树和一条有 $m$ 个点的链。树和链上的每个点均有一个颜色和一个非负点权。

保证树上所有点的颜色两两不同，链上所有点的颜色也两两不同；树与链之间可以出现相同颜色。

你需要在树上选择一条简单路径，并在链上选择一段连续区间。两者都可以为空。要求选择的所有点的颜色两两不同。

求被选择点的点权和的最大值。

$$
n \le 10^5,m \le10^6
$$
## 思路
点分治，考虑所有经过当前重心 $o$ 的路径，设 $h$ 为 $o$ 到 $u$ 路径上的边权和，设 $o \sim u$ 这条路径可以选择的一个链上的区间且不能向两边扩展（因为权值 $>0$），使得没有相同颜色。答案就是 $\max(h+h'-a_u+|[l,r] \cap [l',r']|)$。

这样直接做复杂度肯定是爆炸的，因为一条路径 $o \sim u$ 可能会把链分成 $O(n)$ 个区间。但是我们发现对于从 $u$ 到 $v$，这些区间只会变化 $O(1)$ 个。

那么我们可以考虑对每个区间维护能取到的最大的 $h$。从 $u \to v$，只会把一个区间分成两个小的，回溯的时候再合并成一个。查询最大的 $h$ 可以用 $dfs$ 序，转为区间最小值。区间总数就是 $O(sz_o)$ 的。

记录四元组 $(l,r,h,col)$ 表示 $[l,r]$ 这个区间，能取到的最大的 $h$，它的子树编号是 $col$（因为要经过 $o$，必须是两个不同子树的点）。

将所有四元组按照 $l$ 从小到大排序，考虑当前的 $[l,r,h,col]$ 和另一组 $[l',r',h',col']$ 的贡献。有 $l' < l,col \ne col'$。$s$ 是链上的前缀和数组。

- 若 $l \le r' \le r$，贡献是 $(s_{r'}-s_{l-1})+h+h'-a_o$，查询最大的 $s_{r'}+h'$ 即可。
- 若 $r'>r$，贡献就是 $(s_r-s_{l-1})+h+h'-a_o$。查询最大的 $h'$ 即可。

用树状数组维护最大和其余子树的最大值即可（因为有可能最大值在一颗子树内）。时间复杂度 $O(n\log^2 n+m)$。
## 代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 55e4 + 10, inf = 1e18;
int n, m, c[N], sz[N], vis[N], st[N], tim, f[N][22], col, a[N], b[N], d[N];
int ans = 0, s[N], p[N];
vector<int> E[N];
set<int> q;
struct node {
    int l, r, h, c;
    friend bool operator < (const node &x, const node &y) {
        return x.l < y.l;
    }
};
vector<node> g;
inline void getsz(int u, int fa) {
    sz[u] = 1;
    for (int v : E[u]) if (v != fa && !vis[v])
        getsz(v, u), sz[u] += sz[v];
}
inline int root(int u, int fa, int tot) {
    for (int v : E[u])
        if (v != fa && !vis[v] && sz[v] * 2 > tot) return root(v, u, tot);
    return u;
}
inline void put(int x, int y) {
    if (x > y || st[x] > tim) return;
    int k = __lg(tim - st[x] + 1); 
    int h = max(f[tim][k], f[st[x] + (1 << k) - 1][k]);
    g.push_back({x + 1, y - 1, h, col});
}
inline void change(int x, bool ins) {
    if (!x) return;
    auto it = q.lower_bound(x);
    int l = *prev(it);
    if (ins) {
        put(l, *it), q.insert(it, x);
        st[l] = st[x] = tim + 1;
    }else {
        put(l, x), put(x, *next(it)), q.erase(it);
        st[l] = tim + 1;
    }
}
inline void dfs(int u, int fa, int h) {
    change(c[u], 1);
    f[++ tim][0] = h;
    for (int i = 1; (1 << i) <= tim; i ++)
        f[tim][i] = max(f[tim][i - 1], f[tim - (1 << i - 1)][i - 1]);
    for (int v : E[u]) if (v != fa && !vis[v])
        dfs(v, u, h + a[v]);
    change(c[u], 0);
}
struct info {
    int a = -inf, b = -inf, c = -1;
    inline void add(int v, int k) {
        if (c == k) a = max(a, v);
        else if (v > a) b = a, a = v, c = k;
        else b = max(b, v);
    }
};
struct bit {
    vector<info> t;
    bit(int n) {
        t.resize(n + 1);
    }
    inline void add(int x, int v, int k) {
        for (; x < t.size(); x += x & -x) t[x].add(v, k);
    }
    inline int qry(int x, int k) {
        int ans = -inf;
        for (; x; x = (x - 1) & x) ans = max(ans, t[x].c == k ? t[x].b : t[x].a);
            return ans;
    }
};
inline void work(int u) {
    g.clear(), q.clear();
    q.insert(0), q.insert(m + 2);
    if (c[u]) q.insert(c[u]);
    for (auto it = next(q.begin()); it != q.end(); it ++)
        ans = max(ans, a[u] + s[*it - 1] - s[*prev(it)]);
    for (int v : E[u]) if (!vis[v]) {
        tim = 0, col = v, st[0] = st[c[u]] = 1;
        dfs(v, u, a[u] + a[v]);
        for (auto it = next(q.begin()); it != q.end(); it ++)
            put(*prev(it), *it);
    }
    sort (g.begin(), g.end());
    vector<int> b;
    for (auto &t : g) b.push_back(t.r);
    sort (b.begin(), b.end()), b.erase(unique(b.begin(), b.end()), b.end());
    int k = b.size();
    bit pre(k), suf(k);
    for (auto [l, r, h, id] : g) {
        int x = lower_bound(b.begin(), b.end(), r) - b.begin() + 1;
        ans = max(ans, h + s[r] - s[l - 1]);
        int v = max(pre.qry(x, id), suf.qry(k - x + 1, id) + s[r]);
        ans = max(ans, h + v - a[u] - s[l - 1]);
        pre.add(x, h + s[r], id), suf.add(k - x + 1, h, id);
    }
}
inline void solve(int u) {
    getsz(u, 0), u = root(u, 0, sz[u]);
    vis[u] = 1, work(u);
    for (int v : E[u]) if (!vis[v]) solve(v);
}
signed main() {
   ios :: sync_with_stdio(0);
   cin.tie(0), cout.tie(0);
    cin >> n >> m;
    for (int i = 1, u, v; i < n; i ++) {
        cin >> u >> v;
        E[u].push_back(v), E[v].push_back(u);
    }
    for (int i = 1; i <= n; i ++) cin >> c[i];
    for (int i = 1; i <= n; i ++) cin >> a[i];
    for (int i = 1, x; i <= m; i ++) cin >> x, p[x] = i;
    for (int i = 1; i <= m; i ++) cin >> s[i], s[i] += s[i - 1];
    for (int i = 1; i <= n; i ++) c[i] = p[c[i]];
    s[m + 1] = ans = s[m];
    solve(1);
    cout << ans << '\n';
    return 0;
}
```
