## T1
### 题意
给定一个 $n\times m$ 的方向网格，每个大写字母所在的格子有一个未点亮的光点。

每个光点可以在第 $1$ 到 $10^9$ 秒中任选一秒点亮，从下一秒起，每秒按当前格的方向走一步，永远移动。要求任何一秒移动和点亮结束后，同一格至多有一个已点亮的光点，两个光点同时交换位置合法。

求最多能点亮多少个光点。若 $g=1$，还需输出每个选中光点的初始行、列和点亮时间。
$n, m \le 10^3$
### 思路
考虑用并查集维护联通块，由于点都在网格内部走，所以 $k$ 个点的联通块有 $k$ 条边，也就是一个环。设环的长度为 $L$，连通块内可以点亮的点有 $w$ 个，这个连通块的贡献就是 $\min(L, w)$。因为若 $w > L$，可以在环上每个点放一个点亮的。

接下来考虑如何构造答案。可以求出每个点 $x$ 到环上一个固定点 $P$ 的距离 $p_x$，可以任取 $\min(L, w)$ 个可点亮的点，记 $x$ 点亮的时间为 $t_x$，那么让 $(p_x + t_x) \bmod L$ 取遍 $0 \sim L - 1$。

求 $p_x$ 是容易的，为了方便可以写带权并查集，有 $p_x - p_{fa_x}=1$，时间复杂度 $O(nm\alpha(nm))$。
### 代码
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e6 + 10;
int n, m, g, vis[1010][1010], fa[N], d[N], len[N], cnt[N];
inline int Id(int x, int y) {
	return (x - 1) * m + y;
}
inline int find(int x) {
	if (x == fa[x]) return x;
	int y = fa[x];
	fa[x] = find(fa[x]);
	d[x] += d[y];
	return fa[x];
}
inline void merge(int u, int v) {
	// u -> v
	int x = find(u), y = find(v);
	if (x == y) {
		len[x] = d[v] - d[u] + 1;
		return;
	}
	//a_x - a_y = (a_x - a_u) + (a_u - a_v) + (a_v - a_y) = -d_u + 1 + d_v 
	int w = -d[u] + 1 + d[v];
	fa[x] = y, d[x] = w; 
}
signed main() {
	cin >> n >> m >> g;
	int tot = n * m;
	for (int i = 1; i <= tot; i ++) fa[i] = i, d[i] = 0;
	for (int i = 1; i <= n; i ++) {
		for (int j = 1; j <= m; j ++) {
			char c;
			cin >> c;
			if (c <= 'Z') vis[i][j] = 1, c += 'z' - 'Z';
			int x = i, y = j;
			if (c == 'n') x --;
			if (c == 's') x ++;
			if (c == 'e') y ++;
			if (c == 'w') y --;
//			cout << x << ' ' << y << '\n';
			merge(Id(i, j), Id(x, y));
		}
	}
	vector<array<int, 3> > ans;
	for (int i = 1; i <= n; i ++)
		for (int j = 1; j <= m; j ++) {
			int x = find(Id(i, j));
			if (cnt[x] < len[x] && vis[i][j]) {
				//t_y + d_y = cnt[x] (mod L)
				int t = (cnt[x] - d[Id(i, j)] % len[x] + len[x]) % len[x];
				if (!t) t += len[x];
				ans.push_back({i, j, t});
				cnt[x] ++;
			}
		} 
	cout << ans.size() << '\n';
	if (g) {
		for (auto x : ans) cout << x[0] << ' ' << x[1] << ' ' << x[2] << '\n';
	}
	return 0;
}
```
## T2
### 题意
给定长度为 $n$ 的亮度数组 $h$。每次询问 $[l,r]$，从中任选一个非空子区间。设子区间最大值为 $M$，它第一次和最后一次出现的位置分别为 $A,B$，求 $M(B-A)$ 的最大值。

共 $q$ 次询问，强制在线。第一次区间直接给出，后续区间由上一次的区间和答案生成。最终输出 $\bigoplus_{i=1}^{q}(\mathrm{ans}_i+i)$。
$n \le 5 \times 10^5, q \le 3 \times 10^6$
### 思路
很牛的题。

考虑分治（其实很难想到），对于询问 $l \le mid < r$，如果子区间不跨过 $mid$，可以方便的维护一个前后缀的答案。

接下来考虑跨过 $mid$ 的子区间，若有 $H_1 > H_2$ 且 $H_1,H_2$ 都在 $mid$ 两边出现过，那么 $H_1$ 一定优于 $H_2$。因为 $H_1$ 的位置一定在 $H_2$ 的区间外，所以长度更大。所以只要查询同时在左右两边出现过的最大的数 $H$ 即可。

这个就先把左右两边都出现过的数求出来，从 $mid$ 向外扩展，值一定不减，可以维护出 $R_x$ 表示右端点为 $\le x$，都出现过的最大值，$L_x$ 同理。$[l,r]$ 的答案就是 $H=\min(L_l,R_r)$。至于值 $H$ 的最左最右出现次数，容易线性维护。$O(1)$ 定位到区间也是容易的。

时间复杂度 $O(n\log n+q)$。随便过，不精细实现多只 $\log$ 应该也问题不大。
### 代码
```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 5e5 + 10, K = 19;
int n, q, best[K][N];
int h[N], a[N], b[N], pos[K][N], le[N], ri[N];
inline int qry(int l, int r) {
    if (l == r) return 0;
    int d = 63 - __builtin_clzll((l - 1) ^ (r - 1));
    int ans = max(best[d][l], best[d][r]);
    int x = pos[d][l], y = pos[d][r];
    if (!x || !y) return ans;
    if (h[x] < h[y]) y = ri[x];
    else if (h[x] > h[y]) x = le[y];
    return max(ans, h[x] * (y - x));
}
inline void build(int d, int l, int mid, int r) {
    int s = 0, t = 0, mx = 0;
    for (int i = mid; i >= l; i --) {
        mx = max(mx, h[i]);
        if (h[i] == mx) a[++ s] = i;
    }
    mx = 0;
    for (int i = mid + 1; i <= r; i ++) {
        mx = max(mx, h[i]);
        if (h[i] == mx) b[++ t] = i;
    }
    int i = 1, j = 1;
    while (i <= s && j <= t) {
        if (h[a[i]] < h[b[j]]) {
            i ++;
            continue;
        }
        if (h[a[i]] > h[b[j]]) {
            j ++;
            continue;
        }
        int x = i, y = j, v = h[a[i]];
        while (i <= s && h[a[i]] == v) i ++;
        while (j <= t && h[b[j]] == v) j ++;
        for (int k = x; k < i; k ++) {
            pos[d][a[k]] = a[k];
            ri[a[k]] = b[j - 1];
        }
        for (int k = y; k < j; k ++) {
            pos[d][b[k]] = b[k];
            le[b[k]] = a[i - 1];
        }
    }
    int p = 0;
    for (int k = mid; k >= l; k --) {
        if (pos[d][k]) p = k;
        pos[d][k] = p;
        best[d][k] = qry(k, mid);
    }
    p = 0;
    for (int k = mid + 1; k <= r; k ++) {
        if (pos[d][k]) p = k;
        pos[d][k] = p;
        best[d][k] = qry(mid + 1, k);
    }
}
signed main() {
    ios :: sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    cin >> n >> q;
    for (int i = 1; i <= n; i ++) cin >> h[i];
    for (int d = 0; (1 << d) < n; d ++) {
        int len = 1 << d;
        for (int l = 1; l + len <= n; l += 2 * len)
            build(d, l, l + len - 1, min(n, l + 2 * len - 1));
    }
    int l, r, ans = 0;
    cin >> l >> r;
    for (int i = 1; i <= q; i ++) {
        int z = qry(l, r);
        ans ^= (z + i);
        int x = (l + 37 * z) % n + 1;
        int y = (l + r + 53 * z) % n + 1;
        l = min(x, y), r = max(x, y);
    }
    cout << ans << '\n';
    return 0;
}
```
## T3
### 题意
给定一个至少含有一个 `1` 的 $N\times M$ 的 `01` 矩阵。选定一种 $h\times w$ 的矩形印章，可以平移、重复盖章，允许重叠，但不能旋转、越界或盖到 `0`。求能覆盖所有 `1` 的最大印章面积 $h\times w$。
$N,M \le 2500$。
### 思路
显然可以双指针，$w$ 增大 $h$ 必然不增，考虑如何快速 $check$。
可以维护一个行的最短 $1$ 的长度 $W$，和列的最短 $1$ 的长度 $H$。那么 $w \le W，h \le H$。

对于一个极长的 $1$ 段 $[l,r]$，对于 $x\in [l, r - w+1]$ 时这一行激活。

若激活段 $[u,v]$ 在 $x\in[a,b]$ 期间存在，且 $v-u+1\ge h$，印章就能上下、左右滑动，盖满矩形
$$
[u,v]\times[a,b+w-1]
$$
每次激活或取消一行，会更改 $O(1)$ 个段，在取消的时候记录矩形。

用 set 维护，合法的条件等价于矩形面积并等于 $1$ 的个数。矩形面积并用扫描线即可。

设所有行有 $k$ 个 $1$ 连续段，有 $kW \le NM$，双指针 $W+H=O(W)$ 次（因为 $W < H$ 可以转置），每次复杂度 $O(k \log k)$，总复杂度 $O(nm \log nm)$。由于 set 和扫描线常数过大，需要卡常才能通过。

### 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 2510;
int n, m, H, W, tot, cnt[N << 2], len[N << 2], st[N];
string a[N];
struct run {
    int i, l, r;
};
struct edge {
    int x, l, r, d;
    bool operator < (const edge &b) const {
        return x < b.x;
    }
};
vector<run> seg;
vector<edge> e;
inline void upd(int p, int l, int r, int L, int R, int d) {
    if (L <= l && r <= R) cnt[p] += d;
    else {
        int mid = (l + r) >> 1;
        if (L <= mid) upd(p << 1, l, mid, L, R, d);
        if (R > mid) upd(p << 1 | 1, mid + 1, r, L, R, d);
    }
    if (cnt[p]) len[p] = r - l + 1;
    else if (l == r) len[p] = 0;
    else len[p] = len[p << 1] + len[p << 1 | 1];
}
inline void close_seg(int l, int r, int x, int h, int w) {
    if (r - l + 1 < h || st[l] == x) return;
    e.push_back({st[l], l, r, 1});
    e.push_back({x + w - 1, l, r, -1});
}
inline bool check(int h, int w) {
    if (h == 1 || w == 1) return true;
    vector<pair<int, int>> q;
    q.reserve(2 * seg.size());
    for (auto [i, l, r] : seg) {
        q.emplace_back(l, i);
        q.emplace_back(r - w + 2, -i);
    }
    sort(q.begin(), q.end());
    set<int> s;
    for (int i = 0; i <= n + 1; i ++) s.insert(s.end(), i);
    e.clear();
    for (auto [x, v] : q) {
        int i = abs(v);
        if (v > 0) {
            auto it = s.find(i);
            int l = *prev(it) + 1, r = *next(it) - 1;
            close_seg(l, i - 1, x, h, w);
            close_seg(i + 1, r, x, h, w);
            s.erase(it);
            st[l] = x;
        } else {
            auto it = s.insert(i).first;
            int l = *prev(it) + 1, r = *next(it) - 1;
            close_seg(l, r, x, h, w);
            st[l] = st[i + 1] = x;
        }
    }
    sort(e.begin(), e.end());
    int area = 0, pre = 0;
    for (auto [x, l, r, d] : e) {
        area += (x - pre) * len[1];
        upd(1, 1, n, l, r, d);
        pre = x;
    }
    return area == tot;
}
inline int getw() {
    int res = m;
    for (int i = 1; i <= n; i ++)
        for (int j = 1; j <= m; j ++) {
            if (a[i][j] == '0') continue;
            int l = j;
            while (j < m && a[i][j + 1] == '1') j ++;
            res = min(res, j - l + 1);
        }
    return res;
}
inline void trans() {
    vector<string> b(m + 1, string(n + 1, '0'));
    for (int i = 1; i <= n; i ++)
        for (int j = 1; j <= m; j ++) b[j][i] = a[i][j];
    swap(n, m);
    for (int i = 1; i <= n; i ++) a[i].swap(b[i]);
}
signed main() {
    ios :: sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    cin >> n >> m;
    for (int i = 1; i <= n; i ++) cin >> a[i], a[i] = " " + a[i];
    H = getw();
    trans();
    W = getw();
    if (H > W) trans(), swap(H, W);
    for (int i = 1; i <= n; i ++)
        for (int j = 1; j <= m; j ++) {
            if (a[i][j] == '0') continue;
            int l = j;
            while (j < m && a[i][j + 1] == '1') j ++;
            seg.push_back({i, l, j});
            tot += j - l + 1;
        }
    int ans = W, h = H; 
    for (int w = 2; w <= W && h > 1; w ++) {
        while (h > 1 && !check(h, w)) h --;
        ans = max(ans, h * w);
    }
    cout << ans << '\n';
    return 0;
}
```
## T4
### 题意
给定一个 $n$ 个点 $m$ 条边的无向图，求钦定每个 $i \in [1,n]$ 的点为叶子的最小生成树。
$n, m \le 2 \times 10^5$。
### 思路
钦定 $x$ 为叶子，就将剩下的 $n - 1$ 个点跑最小生成树再加入与 $x$ 相连的最小的边。

先跑出 $n$ 个点的最小生成树 $T$，不与 $x$ 相连的边都保留。

考虑一条非树边 $(u,v,w)$，可能更新的点只有 $T$ 上 $u \to v$ 的路径上的点。对于 $lca(u,v)$，这条边可以连接两颗儿子。否则对于路径上的其他点，都是连接儿子和父亲。

对于一个儿子只要保留边权最小的即可。将非树边按照 $w$ 排序，用并查集维护未覆盖的点。

对于每一个钦定为叶子的 $x$，对所有边跑一遍最小生成树即可。

时间复杂度 $O(m \log n)$。
### 代码
```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e5 + 10, M = 2e5 + 10, K = 17;
int n, m, tot, fa[N][K], dep[N], f[N << 1], nxt[N];
int mi[N], ans[N], cnt[N];
vector<int> g[N];
struct edge {
	int u, v, w;
	bool operator < (const edge &b) const {
		return w < b.w;
	}
} e[M];
bool vis[M];
inline int find(int f[], int x) {
	return f[x] == x ? x : f[x] = find(f, f[x]);
}
inline void dfs(int u, int p) {
	dep[u] = dep[p] + 1, fa[u][0] = p;
	for (int i = 1; i < K; i ++)
		fa[u][i] = fa[fa[u][i - 1]][i - 1];
	for (int v : g[u])
		if (v != p) dfs(v, u);
}
inline int jump(int u, int k) {
	for (int i = 0; k; i ++, k >>= 1)
		if (k & 1) u = fa[u][i];
	return u;
}
inline int lca(int u, int v) {
	if (dep[u] < dep[v]) swap(u, v);
	u = jump(u, dep[u] - dep[v]);
	if (u == v) return u;
	for (int i = K - 1; i >= 0; i --)
		if (fa[u][i] != fa[v][i])
			u = fa[u][i], v = fa[v][i];
	return fa[u][0];
}
inline void add(int u, int x, int y, int w) {
	x = find(f, x), y = find(f, y);
	if (x == y) return;
	f[x] = y, ans[u] += w, cnt[u] --;
}
inline int work(int x, int z, int w) {
	if (x == z) return 0;
	int a = jump(x, dep[x] - dep[z] - 1);
	x = find(nxt, x);
	while (dep[x] > dep[a]) {
		int p = fa[x][0];
		add(p, x, n + p, w);
		x = nxt[x] = find(nxt, p);
	}
	return a;
}
inline void solve() {
	cin >> n >> m;
	tot = 0;
	for (int i = 1; i <= n; i ++) {
		g[i].clear();
		f[i] = nxt[i] = i, mi[i] = LLONG_MAX;
		ans[i] = cnt[i] = 0;
	}
	for (int i = 1; i <= m; i ++) {
		auto &[u, v, w] = e[i];
		cin >> u >> v >> w;
		mi[u] = min(mi[u], w), mi[v] = min(mi[v], w);
		vis[i] = 0;
	}
	sort(e + 1, e + m + 1);
	for (int i = 1; i <= m; i ++) {
		auto [u, v, w] = e[i];
		int x = find(f, u), y = find(f, v);
		if (x == y) continue;
		f[x] = y, vis[i] = 1, tot += w;
		g[u].push_back(v), g[v].push_back(u);
		ans[u] -= w, ans[v] -= w;
		cnt[u] ++, cnt[v] ++;
	}
	dfs(1, 0);
	for (int i = 1; i <= 2 * n; i ++) f[i] = i;
	for (int i = 1; i <= m; i ++) {
		if (vis[i]) continue;
		auto [u, v, w] = e[i];
		int z = lca(u, v);
		int x = work(u, z, w), y = work(v, z, w);
		if (x && y) add(z, x, y, w);
	}
	for (int i = 1; i <= n; i ++)
		cout << (cnt[i] == 1 ? tot + ans[i] + mi[i] : -1) << ' ';
	cout << '\n'; 
}
signed main() {
	ios :: sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	int T;
	cin >> T;
	while (T --) solve();
	return 0;
}
```
