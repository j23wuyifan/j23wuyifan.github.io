## 题意
$T$ 次询问 
$$
\left( \sum_{i=1}^n \sum_{j=1}^m \varphi(ij) \right)\! \bmod 998244353
$$
$n, m \le 10^5, T \le 10^4$
## 思路
简单题。

设 $d = (i,j)$，有 $\varphi(ij)=\varphi(i)\varphi(j)\frac{d}{\varphi(d)}$。

化一下式子：
$$
\begin{aligned}
\sum_{i=1}^n\sum_{j=1}^m\varphi(ij)
&=\sum_{i=1}^n\sum_{j=1}^m\varphi(i)\varphi(j)\frac{d}{\varphi(d)}\\
&= \sum_{d=1}^n\frac{d}{\varphi(d)}\sum_{i=1}^{n/d}\sum_{j=1}^{m/d}[(i,j)=1]\varphi(id)\varphi(jd)\\
&= \sum_{d=1}^n\frac{d}{\varphi(d)}\sum_{p=1}^{n/d}\mu(p)\sum_{i=1}^{n/dp}\sum_{j=1}^{m/dp}\varphi(idp)\varphi(jdp)\\
&= \sum_{d=1}^n\frac{d}{\varphi(d)}\sum_{p=1}^{n/d}\mu(p) 
\left( \sum_{i=1}^{n/dp}\varphi(idp)\right)\left( \sum_{i=1}^{m/dp}\varphi(idp)\right) \\
\end{aligned}
$$
记 $f(n,m)=\sum_{i=1}^n \varphi(im)$。设 $T=dp$。
$$
\begin{aligned}
\sum_{d=1}^n\frac{d}{\varphi(d)}\sum_{p=1}^{n/d}\mu(p) 
\left( \sum_{i=1}^{n/dp}\varphi(idp)\right)\left( \sum_{i=1}^{m/dp}\varphi(idp)\right)
&= \sum_{T=1}^n f(n/T,T)f(m/T,T) \sum_{d |T} \frac{d}{\varphi(d)} \mu(\frac Td)
\end{aligned}
$$
记 $g(T)=\sum_{d |T} \frac{d}{\varphi(d)} \mu(\frac Td)$，这个是容易预处理的。

查询就是 $\sum_{T=1}^n g(T)f(m/T,T)f(n/T,T)$。

对 $T$ 根号分治一下，对于 $T \le \sqrt{n}$ 的暴力求，对于 $T > \sqrt{n}$ 的 $(T,n/T,m/T)$ 的本质不同三元组只有 
$$
\begin{aligned}
\sum_{i=\lceil\sqrt n\rceil}^n\left(\frac ni\right)^2
&=n^2\sum_{i=\lceil\sqrt n\rceil}^n\frac1{i^2}\\
&=\Theta\left(n^2\int_{\sqrt n}^n\frac1{x^2}\,dx\right)\\
&=\Theta\left(n^2\left(\frac1{\sqrt n}-\frac1n\right)\right)\\
&=\Theta(n\sqrt{n})
\end{aligned}
$$
这个直接预处理即可，对相同的 $(n/T,m/T)$ 对 $T$ 做前缀和，查询整除分块。时间复杂度 $O((T+n)\sqrt{n})$。
## 代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e5 + 10, mod = 998244353, B = 5000, K = N / B + 5;
int n, m, TT, phi[N], g[N], mu[N], pr[N], cnt, inv[N];
vector<int> f[N], t[K][K];
bool vis[N];
signed main() {
	n = N - 10;
	mu[1] = 1, phi[1] = 1;
	inv[1] = 1;
	for (int i = 2; i <= n; i ++) {
		inv[i] = inv[mod % i] * (mod - mod / i) % mod;
		if (!vis[i]) pr[++ cnt] = i, mu[i] = -1, phi[i] = i - 1;
		for (int j = 1; j <= cnt && pr[j] * i <= n; j ++) {
			vis[pr[j] * i] = 1;
			if (i % pr[j] == 0) {
				phi[i * pr[j]] = phi[i] * pr[j];
				break;
			}
			mu[i * pr[j]] = -mu[i];
			phi[i * pr[j]] = phi[i] * phi[pr[j]];
		}
	}
	for (int d = 1; d <= n; d ++)
		for (int T = d; T <= n; T += d)
			g[T] = (g[T] + d * inv[phi[d]] % mod * mu[T / d]) % mod;
	for (int i = 1; i <= n; i ++) 
		f[i].resize(n / i + 1);
	for (int T = 1; T <= n; T ++) {
		for (int k = 1; k <= n / T; k ++) {
			f[k][T] = ((k > 1 ? f[k - 1][T] : 0) + phi[k * T]) % mod;
		}
	}
	for (int i = 1; i < K; i ++)
		for (int j = i; j < K; j ++)
			t[i][j].resize(n / i + 2);
	for (int T = B + 1; T <= n; T ++)
		for (int i = 1; i <= n / T; i ++)
			for (int j = i; j <= n / T; j ++)
				t[i][j][T] = f[i][T] * f[j][T] % mod * g[T] % mod;
	for (int i = 1; i < K; i ++)
		for (int j = i; j < K; j ++)
			for (int k = 1; k < t[i][j].size(); k ++)
				t[i][j][k] = (t[i][j][k - 1] + t[i][j][k]) % mod;
	cin >> TT;
	while (TT --) {
		cin >> n >> m;
		if (n > m) swap(n, m);
		int ans = 0;
		for (int i = 1; i <= min(n, B); i ++) 
		ans = (ans + g[i] * f[n / i][i] % mod * f[m / i][i]) % mod;
		for (int l = B + 1, r; l <= n; l = r + 1) {
			r = min(n / (n / l), m / (m / l));
			ans = (ans + t[n / l][m / l][r] - t[n / l][m / l][l - 1]) % mod;
		}
		cout << (ans + mod) % mod << '\n';
	}
	return 0;
}
```
