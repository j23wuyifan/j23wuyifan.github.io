## 题意
给定一片森林，要求至少访问一次所有指定点。可以花 $r_u$ 直接记起 $u$，也可以在记着父亲时花 $t_u$ 联想出 $u$。最多同时记住 $k$ 个点，忘记免费，允许重复记起。
$n \le 5000, k \le 5$。
## 思路
指定点可以不用管。考虑一颗树全选的最小代价。

每记起一次 $u$，至多有一个儿子能用满当前的 $j$ 个位置。 保留 $u$ 时，可以反复联想儿子，但每次只剩 $j-1$ 个位置处理它。最后一次联想出儿子后，可以立刻忘掉 $u$，这个儿子就有 $j$ 个位置。

因此，记起 $u$ 至多 $c$ 次，相当于有 $c$ 次这样的机会分给儿子。

设 $f_{u,j,c}$ 表示子树内可以最多同时记住 $j$ 个数，$u$ 可以免费记起 $c$ 次的最小代价。$g_u$ 表示不靠子树外，选完 $u$ 子树的最小贡献。 

固定 $j\ge2$，对儿子 $v$ 枚举使用的机会数 $d$：
- $d=0$，可以独立处理 $g_v$；或者保留 $u$，联想 $v$ 共 $q$ 次，代价 $t_v \cdot q+f_{v,j-1,q}$。 
- $d>0$，联想 $v$ 共 $d$ 次，代价 $t_v \cdot d+f_{v,j,d}$；或者联想 $d$ 次，再回想 $v$ 共 $q$ 次，代价 $t_v \cdot d+r_v\cdot q+f_{v,j,q+d}$。

$f_{u,j,c}$ 就是所有子树 $\sum d \le c$ 的最小代价。做个多重背包即可。$g_v = \min f_{u,j,c} + r_u \cdot j$。

时间复杂度 $O(n^2k)$。
## 代码

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 5010, inf = 1e18;
int T, n, m, k, p[N], r[N], t[N], a[N], g[N];
vector<int> E[N], f[N][7];
inline void dfs(int u) {
    int s = 0;
    for (int v : E[u]) {
        dfs(v);
        s += g[v];
    }
    g[u] = a[u] ? inf : s;
    f[u][1] = {inf, s};
    for (int j = 2; j <= k; j ++) {
        vector<int> h(1, 0);
        for (int v : E[u]) {
            auto &x = f[v][j - 1], y = f[v][j];
            int len = y.size();
            vector<int> b(len);
            b[0] = g[v];
            for (int q = 1; q < x.size(); q ++) 
                b[0] = min(b[0], q * t[v] + x[q]);
            for (int d = len - 1; d >= 1; d --) {
                b[d] = y[d] + t[v] * d;
                if (d + 1 < len)
                    b[d] = min(b[d], b[d + 1] + r[v] - t[v]);
            }
            for (int d = 1; d < len; d ++) b[d] = min(b[d], b[d - 1]);
            while (len > 1 && b[len - 1] == b[len - 2]) len --;
            vector<int> nw(h.size() + len - 1, inf);
            for (int c = 0; c < h.size(); c ++) 
                for (int d = 0; d < len; d ++)
                    nw[c + d] = min(nw[c + d], h[c] + b[d]);
            h.swap(nw);
        }
        if (h.size() == 1) h.push_back(h[0]);
        h[0] = inf;
        f[u][j].swap(h);
    }
    for (int c = 1; c < f[u][k].size(); c ++)
        g[u] = min(g[u], f[u][k][c] + c * r[u]);
    for (int v : E[u])
        for (int j = 1; j <= k; j ++) vector<int>().swap(f[v][j]);
}
inline void solve() {
    cin >> n >> m >> k;
    memset(a, 0, sizeof(a));
    for (int i = 0; i < N; i ++) E[i].clear();
    for (int i = 1; i <= n; i ++) cin >> p[i], E[p[i]].push_back(i);
    for (int i = 1; i <= n; i ++) cin >> r[i];
    for (int i = 1; i <= n; i ++) cin >> t[i];
    for (int i = 1, x; i <= m; i ++) cin >> x, a[x] = 1;
    int ans = 0;
    for (int x : E[0]) {
        dfs(x);
        ans += g[x];
        for (int j = 1; j <= k; j ++) vector<int>().swap(f[x][j]);
    }
    cout << ans << '\n';
}
signed main() {
    cin >> T;
    while (T --) solve();
    return 0;
}
```
