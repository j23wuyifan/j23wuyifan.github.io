## 题意

给定 $n$ 个顶点的树，顶点编号为 $1,\dots,n$，给定长度 $n_0$ 的序列 $a_1,\dots,a_{n_0}$，共 $m$ 次查询，每次查询给定 $l,r,x$，问树的顶点 $x$，依次向 $a_l,\dots,a_r$ 移动一步，到达的顶点。

若 $x=y$，则从顶点 $x$ 向 $y$ 移动一步到达 $x$，否则到达与 $x$ 在树上相邻且距离 $y$ 最近的位置。强制在线。
$1 \le n,n0,m\le 10^5$。

## 思路
对 $a$ 数组分块，处理出 $f_{u,i}$ 表示 $u$ 这个点经过 $i$ 这个块后变成了啥，查询整块直接暴力跳就是 $O(\frac{n}{B})$ 的。

对于散块，考虑 $x$ 向 $y$ 方向跳一步，有 $2$ 种情况。
- $y$ 不在 $x$ 的子树内，$x \gets fa_x$。
- $y$ 在 $x$ 的子树内，设 $k = dep_y - dep_x$，$x$ 变成 $y$ 的 $k - 1$ 级祖先。

判断 $y$ 是否在 $x$ 的子树内用 dfn 序，求 $k$ 级祖先，长剖即可。容易做到单次 $O(1)$，单次查询复杂度就是 $O(B+\frac nB)$。

考虑如何 $O(n)$ 求出一个块的答案。设快内的点构成虚树 $S$，分成 $2$ 种情况。
- $u$ 在 $S$ 外，设 $k$ 是 $u$ 到 $v(v \in S)$ 内的最小步数，若 $k > B$，$f_u$ 为 $u$ 的 $B$ 级祖先；否则在 $k$ 时刻把 $u$ 加入 $v$。时间复杂度 $O(n)$。
- $u$ 在 $S$ 内，维护虚树上每条链，每次暴力平移每条链，点的改变需要判一下，每次操作只要判 $O(B)$ 条链，共操作 $O(B)$ 次，时间复杂度              $O(B^2)$。

因此一个块预处理的复杂度是 $O(B^2+n)$，$B=\sqrt n$ 最优，时间复杂度 $O(n\sqrt n)$。
## 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10, B = 350, K = B * 2 + 5;
int n, n0, a[N], fa[N], dep[N], dfn[N], id[N], ed[N], tim;
int st[N][17], lg[N], m, son[N], len[N], top[N], up[N], dn[N];
vector<int> E[N];
int f[N / B + 2][N];
inline void dfs(int u) {
    len[u] = 1;
    for (int v : E[u]) {
        dep[v] = dep[u] + 1, st[v][0] = u;
        for (int j = 1; j <= lg[dep[v]]; j ++)
            st[v][j] = st[st[v][j - 1]][j - 1];
        dfs(v);
        if (len[v] + 1 > len[u])
            len[u] = len[v] + 1, son[u] = v;
    }
}
inline void build(int u, int t) {
    top[u] = t, dfn[u] = ++ tim, id[tim] = u;
    if (son[u]) build(son[u], t);
    for (int v : E[u])
        if (v != son[u]) build(v, v);
    ed[u] = tim;
}
inline int anc(int u, int k) {
    if (!k) return u;
    int j = lg[k];
    u = st[u][j], k -= 1 << j;
    int t = top[u], d = dep[u] - dep[t] - k;
    return d >= 0 ? dn[dfn[t] + d] : up[dfn[t] - d];
}
inline bool chk(int u, int v) {
    return dfn[u] <= dfn[v] && dfn[v] <= ed[u];
}
inline int go(int u, int v) {
    if (u == v) return u;
    return chk(u, v) ? anc(v, dep[v] - dep[u] - 1) : fa[u];
}
struct list_t {
    int l, r;
}q[N], buf[N], pre[N];
int nxt[N], cnt[N], nr[N], bel[N], pos[N];
int ev[N], ne[N], key[N];
bool vis[N];
struct edge {
    int u, v, off, len, p;
}e[N];
inline void add(list_t &a, list_t b) {
    if (!b.l) return;
    if (a.l) nxt[a.r] = b.l;
    else a.l = b.l;
    a.r = b.r;
}
inline list_t &get(int u) {
    if (!bel[u]) return q[u];
    edge &t = e[bel[u]];
    int p = t.p + pos[u];
    if (p >= t.len) p -= t.len;
    return buf[t.off + p];
}
inline void solve(int L, int R, int *res) {
    int b = R - L + 1, tot = 0, ec = 0;
    memset(vis, 0, sizeof(vis));
    memset(cnt, 0, sizeof(cnt));
    memset(nxt, 0, sizeof(nxt));
    memset(ev, 0, sizeof(ev));
    vis[1] = 1;
    for (int i = L; i <= R; i ++) vis[a[i]] = 1;
    for (int i = n; i > 1; i --) {
        int u = id[i];
        if (vis[u] || cnt[u]) cnt[fa[u]] ++;
    }
    for (int i = 1; i <= n; i ++) {
        int u = id[i];
        bel[u] = 0;
        if (vis[u] || cnt[u]) {
            nr[u] = u, q[u] = {u, u};
            if (vis[u] || cnt[u] > 1) key[++ tot] = u;
        } else {
            nr[u] = nr[fa[u]];
            int d = dep[u] - dep[nr[u]];
            if (d >= b) res[u] = anc(u, b);
            else ne[u] = ev[d], ev[d] = u;
        }
    }
    for (int i = 2; i <= tot; i ++) {
        int v = key[i], u = fa[v], d = 0;
        ec ++;
        while (!vis[u] && cnt[u] == 1) {
            bel[u] = ec, d ++;
            u = fa[u];
        }
        e[ec] = {u, v, 0, d, 0};
    }
    int off = 1;
    for (int i = 1; i <= ec; i ++) {
        edge &t = e[i];
        t.off = off, off += t.len;
        for (int u = fa[t.v], j = t.len - 1; j >= 0; u = fa[u], j --) {
            pos[u] = j;
            buf[t.off + j] = {u, u};
        }
    }
    for (int i = L; i <= R; i ++) {
        int y = a[i];
        for (int j = 1; j <= tot; j ++) {
            int u = key[j];
            pre[j] = q[u], q[u] = {};
        }
        for (int j = 1; j <= ec; j ++) {
            edge &t = e[j];
            if (!t.len) continue;
            bool down = chk(t.v, y);
            int last = t.p ? t.p - 1 : t.len - 1;
            int p = down ? last : t.p;
            list_t &z = buf[t.off + p];
            add(q[down ? t.v : t.u], z), z = {};
            t.p = down ? last : (t.p + 1 == t.len ? 0 : t.p + 1);
        }
        for (int j = 1; j <= tot; j ++)
            if (pre[j].l) add(get(go(key[j], y)), pre[j]);
        for (int u = ev[i - L + 1]; u; u = ne[u])
            add(get(nr[u]), {u, u});
    }
    for (int u = 1; u <= n; u ++)
        if (nr[u] == u)
            for (int v = get(u).l; v; v = nxt[v])
                res[v] = u;
}
signed main() {
    ios :: sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    cin >> n >> n0 >> m;
    for (int i = 2; i <= n; i ++) cin >> fa[i], E[fa[i]].push_back(i), lg[i] = lg[i >> 1] + 1;
    for (int i = 1; i <= n0; i ++) cin >> a[i];
    dfs(1), build(1, 1);
    for (int u = 1; u <= n; u ++) {
        if (top[u] != u) continue;
        int x = u, y = u;
        for (int j = 0; j < len[u]; j ++) {
            dn[dfn[u] + j] = x, up[dfn[u] + j] = y;
            x = son[x], y = fa[y];
        }
    }
    for (int L = 1, i = 1; L <= n0; L += B, i ++) 
        solve(L, min(L + B - 1, n0), f[i]);
    int ans = 0;
    while (m --) {
        int l, r, x;
        cin >> l >> r >> x;
        l ^= ans, r ^= ans, x ^= ans;
        while (l <= r) {
            if ((l - 1) % B == 0 && min(n0, l + B - 1) <= r)
                x = f[(l - 1) / B + 1][x], l += B;
            else x = go(x, a[l ++]);
        }
        cout << (ans = x) << '\n';
    }
    return 0;
}
```
